       import { readFile, stat, readdir } from 'node:fs/promises';
       import { join } from 'node:path';
       import type { ProjectProfile } from '../types/index.js';

       export async function detectRepoProfile(repoPath: string): Promise<ProjectProfile | undefined> {
         // Check path exists (D-13)
         try { await stat(repoPath); } catch { return undefined; }

         const languages: string[] = [];
         const frameworks: string[] = [];
         let packageManager = 'unknown';
         let testRunner: string | undefined;
         let buildCommand: string | undefined;

         // Check package.json
         const pkgPath = join(repoPath, 'package.json');
         try {
           const pkg = JSON.parse(await readFile(pkgPath, 'utf-8'));
           languages.push('typescript'); // refine: check tsconfig.json
           const allDeps = { ...pkg.dependencies, ...pkg.devDependencies };
           if (allDeps['react']) frameworks.push('react');
           if (allDeps['next']) frameworks.push('next.js');
           if (allDeps['express']) frameworks.push('express');
           if (allDeps['hono']) frameworks.push('hono');
           // Package manager detection
           if (pkg.packageManager?.startsWith('pnpm')) packageManager = 'pnpm';
           else if (await fileExists(join(repoPath, 'pnpm-lock.yaml'))) packageManager = 'pnpm';
           else if (await fileExists(join(repoPath, 'yarn.lock'))) packageManager = 'yarn';
           else packageManager = 'npm';
           // Test runner
           if (allDeps['vitest']) testRunner = 'vitest';
           else if (allDeps['jest']) testRunner = 'jest';
           // Build
           if (pkg.scripts?.build) buildCommand = `${packageManager} run build`;
         } catch { /* no package.json */ }

         // Check pom.xml
         if (await fileExists(join(repoPath, 'pom.xml'))) {
           languages.push('java');
           packageManager = 'maven';
         }

         // Check requirements.txt / pyproject.toml
         if (await fileExists(join(repoPath, 'requirements.txt')) || await fileExists(join(repoPath, 'pyproject.toml'))) {
           languages.push('python');
           if (await fileExists(join(repoPath, 'pyproject.toml'))) packageManager = 'poetry';
           else packageManager = 'pip';
         }

         if (languages.length === 0) languages.push('unknown');

         return { languages, frameworks, packageManager, testRunner, buildCommand };
       }

       export async function detectRepoProvides(repoPath: string): Promise<string[]> {
         try { await stat(repoPath); } catch { return []; }
         const provides: string[] = [];

         // Check for pnpm workspaces (D-02: monorepo support)
         if (await fileExists(join(repoPath, 'pnpm-workspace.yaml'))) {
           // Read workspace packages and get their names
           const workspaceContent = await readFile(join(repoPath, 'pnpm-workspace.yaml'), 'utf-8');
           // Simple parsing: extract package dirs, read each package.json
           const packageDirs = parseWorkspacePackages(workspaceContent, repoPath);
           for (const dir of packageDirs) {
             const name = await getPackageName(dir);
             if (name) provides.push(name);
           }
           if (provides.length > 0) return provides;
         }

         // Single package.json
         const name = await getPackageName(repoPath);
         if (name) { provides.push(name); return provides; }

         // pom.xml artifactId
         const pomPath = join(repoPath, 'pom.xml');
         try {
           const pom = await readFile(pomPath, 'utf-8');
           const match = pom.match(/<artifactId>([^<]+)<\/artifactId>/);
           if (match) provides.push(match[1]);
         } catch { /* no pom */ }

         return provides;
       }

       function parseWorkspacePackages(yaml: string, rootPath: string): string[] { ... }
