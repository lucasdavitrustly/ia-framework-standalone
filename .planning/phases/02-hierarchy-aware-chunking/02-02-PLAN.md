1. For each result chunk that has a `superclass` field and `expand_hops >= 1`:
   - Query Qdrant for chunks where `payload.name == chunk.superclass` AND `payload.chunkType == "class"`
   - Add found chunks to results (deduplicated by chunkId)
   - If `expand_hops >= 2` and the superclass chunk also has a `superclass`, fetch that too (recursive up to hops limit)

2. Implementation approach — add a function `expandHierarchy`:

```typescript
async function expandHierarchy(
  results: SearchResult[],
  store: QdrantVectorStore,
  collection: string,
  maxHops: number,
): Promise<SearchResult[]> {
  if (maxHops <= 0) return results;

  const existingIds = new Set(results.map((r) => r.id));
  const toExpand = results.filter((r) => r.payload.superclass && r.payload.chunkType === 'class');
  const expanded: SearchResult[] = [];

  for (const result of toExpand) {
    let current = result.payload.superclass;
    let hopsLeft = maxHops;

    while (current && hopsLeft > 0) {
      // Use scrollChunks with filter: name == current, chunkType == "class"
      const parents = await store.scrollChunks(collection, {
        filter: {
          must: [
            { key: 'name', match: { value: current } },
            { key: 'chunkType', match: { value: 'class' } },
          ],
        },
        limit: 5,
      });

      for (const p of parents.points) {
        if (!existingIds.has(p.id)) {
          existingIds.add(p.id);
          expanded.push({ id: p.id, score: 0, payload: p.payload, expansionSource: 'hierarchy' });
        }
        // Continue up the chain
        current = p.payload.superclass;
      }

      if (parents.points.length === 0) break;
      hopsLeft--;
    }
  }

  return [...results, ...expanded];
}
```
