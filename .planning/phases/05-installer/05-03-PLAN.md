    **scripts/configure-opencode.sh**: Function `configure_opencode()`:

    ```bash
    configure_opencode() {
      local ia_framework_dir="$1"       # absolute path to ia-framework repo
      local config_path="$2"           # absolute path to .ia-framework/project.json
      
      # Determine OpenCode config location
      local opencode_config_dir="${XDG_CONFIG_HOME:-$HOME/.config}/opencode"
      local opencode_config="$opencode_config_dir/config.json"
      
      mkdir -p "$opencode_config_dir"
      
      # Build the MCP server path
      local mcp_bin="$ia_framework_dir/packages/mcp-server/dist/bin.js"
      
      # Check if opencode config exists
      if [ -f "$opencode_config" ]; then
        # Merge: add ia-framework MCP server entry if not already present
        if grep -q '"ia-framework"' "$opencode_config" 2>/dev/null; then
          echo -e "${GREEN}✓ OpenCode MCP already configured${NC}"
          return 0
        fi
        # Use node to merge JSON safely
        node -e "
          const fs = require('fs');
          const config = JSON.parse(fs.readFileSync('$opencode_config', 'utf8'));
          config.mcpServers = config.mcpServers || {};
          config.mcpServers['ia-framework'] = {
            command: 'node',
            args: ['$mcp_bin'],
            env: { IA_FRAMEWORK_CONFIG: '$config_path' }
          };
          fs.writeFileSync('$opencode_config', JSON.stringify(config, null, 2) + '\n');
        "
      else
        # Create new config file
        cat > "$opencode_config" <<EOF
    {
      "mcpServers": {
        "ia-framework": {
          "command": "node",
          "args": ["$mcp_bin"],
          "env": {
            "IA_FRAMEWORK_CONFIG": "$config_path"
          }
        }
      }
    }
    EOF
      fi
      
      echo -e "${GREEN}✓ OpenCode MCP configured: $opencode_config${NC}"
      echo "  Restart OpenCode to load the ia-framework MCP server."
    }
    ```

    - How to verify the MCP server is working (run `opencode` and check MCP tools)
    - Environment variable reference (IA_FRAMEWORK_CONFIG)

    Make the script executable.
