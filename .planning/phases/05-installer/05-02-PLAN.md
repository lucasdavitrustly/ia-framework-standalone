<objective>
Create the `packages/cli` package — a `commander`-based CLI that wraps the framework's core operations.
The CLI exposes three commands for post-installation use:
- `ia-framework index` — index (or re-index) the configured target project
- `ia-framework search <query>` — search the indexed codebase  
- `ia-framework status` — show indexing status (collection info, chunk count, last indexed commit)

Purpose: Lets developers use the framework without re-running the full installer.
Output: `@ia-framework/cli` package with `ia-framework` binary.
</objective>

    ```typescript
    import { Command } from 'commander';
    import { join } from 'node:path';
    import { loadConfig, createLogger } from '@ia-framework/core';
    import { createEmbeddingProvider } from '@ia-framework/embedding';
    import { QdrantVectorStore } from '@ia-framework/vector-store';
    import { IndexingPipeline } from '@ia-framework/indexer';

    export function indexCommand(): Command {
      return new Command('index')
        .description('Index the configured target project codebase into Qdrant')
        .option('-f, --force', 'Force re-index all files regardless of changes', false)
        .option('-c, --config <path>', 'Path to project.json config', '')
        .action(async (options) => {
          const logger = createLogger('cli:index');
          const configPath = options.config || 
            process.env['IA_FRAMEWORK_CONFIG'] || 
            join(process.cwd(), '.ia-framework', 'project.json');
          
          try {
            const config = await loadConfig(configPath);
            const embeddingProvider = createEmbeddingProvider(config.embedding);
            const vectorStore = new QdrantVectorStore({
              host: config.vectorStore.host,
              port: config.vectorStore.port,
              collectionName: config.vectorStore.collectionName,
              apiKey: config.vectorStore.apiKey,
            });
            
            const pipeline = new IndexingPipeline({
              embeddingProvider,
              vectorStore,
              targetProjectPath: config.targetProjectPath,
              indexingConfig: config.indexing,
              logger,
            });
            
            console.log(`Indexing ${config.targetProjectPath}...`);
            const result = await pipeline.run((progress) => {
              process.stdout.write(`\r  ${progress.filesProcessed}/${progress.totalFiles} files — ${progress.chunksIndexed} chunks`);
            });
            console.log(`\n✓ Indexed ${result.chunksIndexed} chunks from ${result.filesProcessed} files`);
          } catch (error) {
            console.error(`✗ Indexing failed: ${String(error)}`);
            process.exit(1);
          }
        });
    }
    ```

    **packages/cli/src/commands/search.command.ts**:
    ```typescript
    import { Command } from 'commander';
    import { join } from 'node:path';
    import { loadConfig, createLogger } from '@ia-framework/core';
    import { createEmbeddingProvider } from '@ia-framework/embedding';
    import { QdrantVectorStore } from '@ia-framework/vector-store';
    import { SearchEngine } from '@ia-framework/retrieval';

    export function searchCommand(): Command {
      return new Command('search')
        .description('Search the indexed codebase with a natural language query')
        .argument('<query>', 'Natural language search query')
        .option('-l, --limit <n>', 'Maximum number of results', '10')
        .option('-t, --threshold <n>', 'Minimum score threshold (0-1)', '0.5')
        .option('-c, --config <path>', 'Path to project.json config', '')
        .action(async (query: string, options) => {
          const logger = createLogger('cli:search');
          const configPath = options.config || 
            process.env['IA_FRAMEWORK_CONFIG'] || 
            join(process.cwd(), '.ia-framework', 'project.json');
          
          try {
            const config = await loadConfig(configPath);
            const embeddingProvider = createEmbeddingProvider(config.embedding);
            const vectorStore = new QdrantVectorStore({
              host: config.vectorStore.host,
              port: config.vectorStore.port,
              collectionName: config.vectorStore.collectionName,
              apiKey: config.vectorStore.apiKey,
            });
            
            const searchEngine = new SearchEngine({ embeddingProvider, vectorStore });
            const results = await searchEngine.search({
              text: query,
              limit: parseInt(options.limit, 10),
              scoreThreshold: parseFloat(options.threshold),
            });
            
            if (results.length === 0) {
              console.log('No results found.');
              return;
            }
            
            console.log(`Found ${results.length} result(s) for: "${query}"\n`);
            for (const result of results) {
              console.log(`─── ${result.relativePath} (${result.chunkType}: ${result.name ?? 'unnamed'}) [score: ${result.score.toFixed(3)}]`);
              console.log(result.content.slice(0, 300) + (result.content.length > 300 ? '...' : ''));
              console.log('');
            }
          } catch (error) {
            console.error(`✗ Search failed: ${String(error)}`);
            process.exit(1);
          }
        });
    }
    ```

    **packages/cli/src/commands/status.command.ts**:
    ```typescript
    import { Command } from 'commander';
    import { join } from 'node:path';
    import { loadConfig } from '@ia-framework/core';
    import { QdrantVectorStore } from '@ia-framework/vector-store';
    import { getCurrentCommit, getLastIndexedCommit } from '@ia-framework/indexer';

    export function statusCommand(): Command {
      return new Command('status')
        .description('Show indexing status and Qdrant collection info')
        .option('-c, --config <path>', 'Path to project.json config', '')
        .action(async (options) => {
          const configPath = options.config || 
            process.env['IA_FRAMEWORK_CONFIG'] || 
            join(process.cwd(), '.ia-framework', 'project.json');
          
          try {
            const config = await loadConfig(configPath);
            
            console.log('ia-framework status');
            console.log('═══════════════════');
            console.log(`Target project: ${config.targetProjectPath}`);
            console.log(`Embedding:      ${config.embedding.provider} / ${config.embedding.model}`);
            console.log(`Vector store:   ${config.vectorStore.host}:${config.vectorStore.port}`);
            
            // Try to connect to Qdrant and get collection info
            try {
              const vectorStore = new QdrantVectorStore({
                host: config.vectorStore.host,
                port: config.vectorStore.port,
                collectionName: config.vectorStore.collectionName,
                apiKey: config.vectorStore.apiKey,
              });
              
              const info = await vectorStore.getCollectionInfo();
              console.log(`\nCollection:     ${info.name}`);
              console.log(`Chunks indexed: ${info.pointsCount ?? 0}`);
            } catch {
              console.log('\nQdrant:         ✗ not reachable (is Docker running?)');
            }
            
            // Show last indexed commit
            try {
              const lastCommit = await getLastIndexedCommit(config.targetProjectPath);
              const currentCommit = await getCurrentCommit(config.targetProjectPath);
              if (lastCommit) {
                const upToDate = lastCommit === currentCommit;
                console.log(`Last indexed:   ${lastCommit.slice(0, 8)} ${upToDate ? '✓ up to date' : '⚠ stale (run ia-framework index)'}`);
              } else {
                console.log(`Last indexed:   never (run ia-framework index)`);
              }
            } catch {
              console.log(`Last indexed:   unknown`);
            }
          } catch (error) {
            console.error(`✗ Status failed: ${String(error)}`);
            process.exit(1);
          }
        });
    }
    ```
