```typescript
interface VectorPoint {
  /** Deterministic chunk ID (from ast-parser) */
  id: string;
  /** Text embedding vector (from text embedding model) */
  textVector: number[];
  /** Code embedding vector (from code embedding model — same or different model) */
  codeVector: number[];
  /** Chunk metadata stored as Qdrant payload */
  payload: ChunkPayload;
}

interface ChunkPayload {
  filePath: string;
  relativePath: string;
  language: string;
  chunkType: string;
  name?: string;
  content: string;
  startLine: number;
  endLine: number;
  imports: string[];
  exports: string[];
  indexedAt: string;
}

interface SearchOptions {
  /** Which named vector to search against ('text' or 'code') */
  vectorName: 'text' | 'code';
  /** Max results to return */
  limit: number;
  /** Minimum similarity score (0-1) */
  scoreThreshold?: number;
  /** Filter by payload fields (Qdrant filter syntax) */
  filter?: Record<string, unknown>;
}

interface SearchResult {
  id: string;
  score: number;
  payload: ChunkPayload;
  vectorName: 'text' | 'code';
}

interface CollectionInfo {
  name: string;
  pointCount: number;
  vectorDimensions: { text: number; code: number };
  status: string;
}

interface VectorStore {
  initialize(config: { textDimensions: number; codeDimensions: number }): Promise<void>;
  upsertBatch(points: VectorPoint[]): Promise<void>;
  search(queryVector: number[], options: SearchOptions): Promise<SearchResult[]>;
  getCollectionInfo(): Promise<CollectionInfo>;
  deleteByFilter(filter: Record<string, unknown>): Promise<number>;
  deleteByIds(ids: string[]): Promise<void>;
}
```

<!-- Qdrant dual-vector collection config -->
Named vectors in Qdrant:
```json
{
  "vectors": {
    "text": { "size": 1536, "distance": "Cosine" },
    "code": { "size": 1536, "distance": "Cosine" }
  }
}
```
Both vectors can use same dimensions (from config). For dual-model indexing: text model and code model may have different dimensions — store both.

<!-- @qdrant/js-client-rest usage -->
- QdrantClient constructor: `new QdrantClient({ host, port })`
- Create collection: `client.createCollection(name, { vectors: {...} })`
- Check exists: `client.collectionExists(name)` → boolean
- Upsert: `client.upsert(name, { wait: true, points: [...] })`
- Search: `client.search(name, { vector: { name: 'text', vector: [...] }, limit, score_threshold, filter, with_payload: true })`
- Collection info: `client.getCollection(name)` → points_count, status
- Delete by filter: `client.delete(name, { wait: true, filter: {...} })`
- Delete by IDs: `client.delete(name, { wait: true, points: [...] })`
- Point ID in Qdrant must be UUID or unsigned integer. Use UUID v5 (namespace + chunk ID string) for deterministic IDs.

<!-- ID handling -->
Qdrant requires UUID or uint64 point IDs. Convert chunk string IDs to UUID v5:
- Use crypto.createHash('sha256') to get 128 bits, then format as UUID
- Or use a simple numeric hash. Keep a mapping in payload.
- Store original string ID in payload.id field.
- Easiest: store the sha1 chunk ID in payload, use UUID v5 from the same input for Qdrant point ID.

    ```typescript
      chunkType: string;

      id: string;

      filter?: Record<string, unknown> | undefined;

      id: string;
      /** Similarity score (0-1) */
      score: number;

      payload: ChunkPayload;

      vectorName: 'text' | 'code';

     * Provider-agnostic vector storage interface.

      initialize(config: { textDimensions: number; codeDimensions: number }): Promise<void>;

      upsertBatch(points: VectorPoint[]): Promise<void>;

      deleteByFilter(filter: Record<string, unknown>): Promise<number>;

      deleteByIds(ids: string[]): Promise<void>;
    ```

    ```typescript
    import { QdrantClient } from '@qdrant/js-client-rest';
    import { createHash } from 'node:crypto';
    import type {
      VectorStore,
      VectorPoint,
      SearchOptions,
      SearchResult,
      CollectionInfo,
      ChunkPayload,
    } from './interface.js';

    /**
     * Convert a string chunk ID to a UUID-format string for Qdrant.
     * Qdrant requires UUID or uint64 point IDs.
     * Uses first 32 hex chars of sha256(input), formatted as UUID.
     */
    function toQdrantId(chunkId: string): string {
      const hash = createHash('sha256').update(chunkId).digest('hex');
      // Format as UUID v4-like: 8-4-4-4-12
      return [
        hash.slice(0, 8),
        hash.slice(8, 12),
        hash.slice(12, 16),
        hash.slice(16, 20),
        hash.slice(20, 32),
      ].join('-');
    }

    export interface QdrantVectorStoreConfig {
      host: string;
      port: number;
      collectionName: string;
      apiKey?: string | undefined;
    }

    export class QdrantVectorStore implements VectorStore {
      private readonly client: QdrantClient;
      private readonly collectionName: string;

      constructor(config: QdrantVectorStoreConfig) {
        this.collectionName = config.collectionName;
        this.client = new QdrantClient({
          host: config.host,
          port: config.port,
          ...(config.apiKey ? { apiKey: config.apiKey } : {}),
        });
      }

      async initialize(config: { textDimensions: number; codeDimensions: number }): Promise<void> {
        const exists = await this.client.collectionExists(this.collectionName);
        if (exists) return;

        await this.client.createCollection(this.collectionName, {
          vectors: {
            text: { size: config.textDimensions, distance: 'Cosine' },
            code: { size: config.codeDimensions, distance: 'Cosine' },
          },
        });
      }

      async upsertBatch(points: VectorPoint[]): Promise<void> {
        if (points.length === 0) return;

        await this.client.upsert(this.collectionName, {
          wait: true,
          points: points.map((p) => ({
            id: toQdrantId(p.id),
            vector: {
              text: p.textVector,
              code: p.codeVector,
            },
            payload: {
              ...p.payload,
            },
          })),
        });
      }

      async search(queryVector: number[], options: SearchOptions): Promise<SearchResult[]> {
        const results = await this.client.search(this.collectionName, {
          vector: { name: options.vectorName, vector: queryVector },
          limit: options.limit,
          score_threshold: options.scoreThreshold,
          filter: options.filter as Parameters<QdrantClient['search']>[1]['filter'],
          with_payload: true,
        });

        return results.map((r) => ({
          id: (r.payload as ChunkPayload).chunkId ?? String(r.id),
          score: r.score,
          payload: r.payload as ChunkPayload,
          vectorName: options.vectorName,
        }));
      }

      async getCollectionInfo(): Promise<CollectionInfo> {
        const info = await this.client.getCollection(this.collectionName);
        const vectors = info.config?.params?.vectors;

        let textDim = 0;
        let codeDim = 0;
        if (vectors && typeof vectors === 'object' && !Array.isArray(vectors)) {
          const textVec = (vectors as Record<string, { size?: number }>)['text'];
          const codeVec = (vectors as Record<string, { size?: number }>)['code'];
          textDim = textVec?.size ?? 0;
          codeDim = codeVec?.size ?? 0;
        }

        return {
          name: this.collectionName,
          pointCount: info.points_count ?? 0,
          vectorDimensions: { text: textDim, code: codeDim },
          status: String(info.status ?? 'unknown'),
        };
      }

      async deleteByFilter(filter: Record<string, unknown>): Promise<number> {
        const result = await this.client.delete(this.collectionName, {
          wait: true,
          filter: filter as Parameters<QdrantClient['delete']>[1]['filter'],
        });
        return (result as { deleted_count?: number }).deleted_count ?? 0;
      }

      async deleteByIds(ids: string[]): Promise<void> {
        if (ids.length === 0) return;
        await this.client.delete(this.collectionName, {
          wait: true,
          points: ids.map(toQdrantId),
        });
      }
    }
    ```
