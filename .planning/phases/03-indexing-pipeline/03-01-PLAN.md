<!-- From @ia-framework/core -->
```typescript
// ChunkMetadata from @ia-framework/core (what gets stored in Qdrant)
interface ChunkMetadata {
  id: string;
  filePath: string;
  relativePath: string;
  language: string;
  chunkType: 'function' | 'class' | 'interface' | 'type' | 'enum' | 'module' | 'other';
  name?: string | undefined;
  content: string;
  startLine: number;
  endLine: number;
  imports: string[];
  exports: string[];
  indexedAt: string;
}
```

<!-- AST Parser's own CodeChunk (subset of ChunkMetadata, without indexedAt) -->
This package produces CodeChunk objects that map 1:1 to ChunkMetadata. The indexer adds indexedAt before storage.

<!-- Tree-sitter approach -->
- Use `tree-sitter` npm package (native Node.js bindings, NOT web-tree-sitter WASM)
- Use `tree-sitter-typescript` for TypeScript/JavaScript/TSX/JSX parsing
- Walk the AST top-down, extracting top-level declarations as chunks:
  - function_declaration, arrow_function (when exported)
  - class_declaration
  - interface_declaration
  - type_alias_declaration
  - enum_declaration
  - export_statement wrapping any of the above
- Import extraction: scan import_statement nodes for source module strings
- Export extraction: collect export_statement subjects (names, not types)
- If a file has no top-level declarations, create one 'module' chunk for the whole file

<!-- Chunking strategy -->
- Max chunk content: 200 lines (hard limit). If a class exceeds 200 lines, split at method boundaries.
- Each chunk must contain complete code units (never split mid-function)
- ChunkID: `sha1(filePath + ':' + startLine + ':' + endLine)` (deterministic for incremental re-indexing)

    ```typescript
    import Parser from 'tree-sitter';
    import TypeScript from 'tree-sitter-typescript';
    import type { CodeChunk } from '../types.js';
    import { createHash } from 'node:crypto';

    // Shared parser instance (tree-sitter parsers are not thread-safe but Node.js is single-threaded)
    const parser = new Parser();

    function makeChunkId(filePath: string, startLine: number, endLine: number): string {
      return createHash('sha1').update(`${filePath}:${startLine}:${endLine}`).digest('hex');
    }

    // Node types that represent top-level declarations
    const DECLARATION_TYPES = new Set([
      'function_declaration',
      'class_declaration',
      'interface_declaration',
      'type_alias_declaration',
      'enum_declaration',
      'abstract_class_declaration',
      'method_definition',
      'lexical_declaration', // const/let at module level
      'variable_statement',  // var at module level
    ]);

    const CHUNK_TYPE_MAP: Record<string, CodeChunk['chunkType']> = {
      function_declaration: 'function',
      class_declaration: 'class',
      abstract_class_declaration: 'class',
      interface_declaration: 'interface',
      type_alias_declaration: 'type',
      enum_declaration: 'enum',
    };

    function getNodeName(node: Parser.SyntaxNode): string | undefined {
      const nameNode = node.childForFieldName('name');
      return nameNode?.text;
    }

    function extractImports(rootNode: Parser.SyntaxNode): string[] {
      const imports: string[] = [];
      for (const child of rootNode.children) {
        if (child.type === 'import_statement') {
          const source = child.childForFieldName('source');
          if (source) {
            // Remove quotes
            imports.push(source.text.slice(1, -1));
          }
        }
      }
      return imports;
    }

    function extractExports(rootNode: Parser.SyntaxNode): string[] {
      const exports: string[] = [];
      for (const child of rootNode.children) {
        if (child.type === 'export_statement') {
          const declaration = child.childForFieldName('declaration');
          if (declaration) {
            const name = getNodeName(declaration);
            if (name) exports.push(name);
          }
        }
      }
      return exports;
    }

    export function parseTypeScript(
      sourceCode: string,
      filePath: string,
      relativePath: string,
      language: string,
      maxChunkLines = 200,
    ): CodeChunk[] {
      // Set language grammar
      const grammar = language === 'tsx' ? TypeScript.tsx : TypeScript.typescript;
      parser.setLanguage(grammar as unknown as Parser.Language);

      const tree = parser.parse(sourceCode);
      const root = tree.rootNode;
      const lines = sourceCode.split('\n');

      const allImports = extractImports(root);
      const chunks: CodeChunk[] = [];
      let hasDeclarations = false;

      for (const child of root.children) {
        let targetNode = child;
        let isExported = false;

        // Unwrap export_statement
        if (child.type === 'export_statement') {
          const declaration = child.childForFieldName('declaration');
          if (declaration) {
            targetNode = declaration;
            isExported = true;
          } else {
            // export { name } or export default expr
            continue;
          }
        }

        if (!DECLARATION_TYPES.has(targetNode.type)) continue;

        hasDeclarations = true;
        const startLine = child.startPosition.row + 1;
        const endLine = child.endPosition.row + 1;
        const content = lines.slice(startLine - 1, endLine).join('\n');

        const chunkType: CodeChunk['chunkType'] = CHUNK_TYPE_MAP[targetNode.type] ?? 'other';
        const name = getNodeName(targetNode);
        const chunkExports = isExported && name ? [name] : [];

        chunks.push({
          id: makeChunkId(filePath, startLine, endLine),
          filePath,
          relativePath,
          language,
          chunkType,
          name,
          content,
          startLine,
          endLine,
          imports: allImports,
          exports: chunkExports,
        });
      }

      // If no top-level declarations, create a module chunk for the whole file
      if (!hasDeclarations) {
        chunks.push({
          id: makeChunkId(filePath, 1, lines.length),
          filePath,
          relativePath,
          language,
          chunkType: 'module',
          name: undefined,
          content: sourceCode,
          startLine: 1,
          endLine: lines.length,
          imports: allImports,
          exports: [],
        });
      }

      return chunks;
    }
    ```

    ```typescript
    import { readFile, stat } from 'node:fs/promises';
    import { extname, relative, resolve } from 'node:path';
    import type { CodeChunk, ChunkerConfig } from './types.js';
    import { parseTypeScript } from './parsers/typescript.js';

    const LANGUAGE_MAP: Record<string, string> = {
      '.ts': 'typescript',
      '.tsx': 'tsx',
      '.js': 'javascript',
      '.jsx': 'jsx',
      '.mjs': 'javascript',
      '.cjs': 'javascript',
    };

    export type { CodeChunk, ChunkerConfig };

    export async function parseFile(
      filePath: string,
      config: ChunkerConfig = {},
    ): Promise<CodeChunk[]> {
      // Verify file exists
      await stat(filePath); // throws if missing

      const ext = extname(filePath).toLowerCase();
      const language = LANGUAGE_MAP[ext];
      if (!language) {
        // Unsupported language — return single module chunk
        const content = await readFile(filePath, 'utf-8');
        const lines = content.split('\n');
        const { createHash } = await import('node:crypto');
        return [{
          id: createHash('sha1').update(`${filePath}:1:${lines.length}`).digest('hex'),
          filePath,
          relativePath: config.projectRoot ? relative(config.projectRoot, filePath) : filePath,
          language: ext.slice(1) || 'unknown',
          chunkType: 'module',
          content,
          startLine: 1,
          endLine: lines.length,
          imports: [],
          exports: [],
        }];
      }

      const content = await readFile(filePath, 'utf-8');
      const relativePath = config.projectRoot ? relative(config.projectRoot, filePath) : filePath;

      // Normalize language for parser
      const parserLanguage = (language === 'tsx' || language === 'jsx') ? language : 
                              (language === 'javascript') ? 'tsx' : 'typescript';

      return parseTypeScript(
        content,
        filePath,
        relativePath,
        language,
        config.maxChunkLines ?? 200,
      );
    }
    ```
