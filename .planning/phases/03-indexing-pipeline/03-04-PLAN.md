- Skip directories: node_modules, .git, dist, build, .next, out, __pycache__, .venv, venv, target, .turbo, coverage

- Default exclude: *.min.js, *.generated.*, *.d.ts, __snapshots__/**, fixtures/**

<!-- Pipeline stages -->
1. FileWalker: discover all source files matching criteria
2. ASTParser: parse each file → CodeChunk[]
3. EmbeddingGenerator: batch embed chunks (BOTH text and code vectors using same provider)
4. QdrantUpserter: batch upsert VectorPoint[] to Qdrant

Note on dual vectors: For Phase 3, use the SAME embedding provider for both text and code vectors (simpler first implementation). The vector-store supports different dimensions for text/code, but initially use the same model for both. Phase 7 benchmarks will evaluate whether a code-specialized model improves results.

    ```typescript
    import { readFile, access } from 'node:fs/promises';
    import { join, extname } from 'node:path';
    import { glob } from 'tinyglobby';
    import ignore from 'ignore';

    const SKIP_DIRS = [
      'node_modules', '.git', 'dist', 'build', '.next', 'out',
      '__pycache__', '.venv', 'venv', 'target', '.turbo', 'coverage',
      '.cache', '.parcel-cache', 'storybook-static',
    ];

    const SOURCE_EXTENSIONS = new Set([
      '.ts', '.tsx', '.js', '.jsx', '.mjs', '.cjs',
      '.py', '.go', '.rs', '.java', '.kt', '.rb',
    ]);

    const DEFAULT_EXCLUDE = [
      '**/*.min.js',
      '**/*.generated.*',
      '**/*.d.ts',
      '**/__snapshots__/**',
      '**/fixtures/**',
      '**/*.test.*',
      '**/*.spec.*',
    ];

    export interface WalkOptions {
      projectPath: string;
      includePatterns?: string[];
      excludePatterns?: string[];
      respectGitignore?: boolean;
    }

    async function loadGitignore(projectPath: string): Promise<ReturnType<typeof ignore> | null> {
      const gitignorePath = join(projectPath, '.gitignore');
      try {
        const content = await readFile(gitignorePath, 'utf-8');
        return ignore().add(content);
      } catch {
        return null;
      }
    }

    export async function walkFiles(options: WalkOptions): Promise<string[]> {
      const {
        projectPath,
        includePatterns,
        excludePatterns = [],
        respectGitignore = true,
      } = options;

      // Build gitignore filter
      const gitignoreFilter = respectGitignore ? await loadGitignore(projectPath) : null;

      // Build skip patterns for tinyglobby
      const ignorePatterns = [
        ...SKIP_DIRS.map((d) => `**/${d}/**`),
        ...DEFAULT_EXCLUDE,
        ...excludePatterns,
      ];

      // Default include: all source files
      const patterns = includePatterns ?? ['**/*'];

      const files = await glob(patterns, {
        cwd: projectPath,
        ignore: ignorePatterns,
        absolute: true,
        onlyFiles: true,
      });

      return files.filter((file) => {
        // Must be a recognized source extension (unless explicit include patterns given)
        if (!includePatterns) {
          const ext = extname(file).toLowerCase();
          if (!SOURCE_EXTENSIONS.has(ext)) return false;
        }

        // Apply gitignore filter
        if (gitignoreFilter) {
          const relativePath = file.slice(projectPath.length + 1);
          if (gitignoreFilter.ignores(relativePath)) return false;
        }

        return true;
      });
    }
    ```

    ```typescript
    import { readFile, writeFile, mkdir } from 'node:fs/promises';
    import { join } from 'node:path';
    import { execFile } from 'node:child_process';
    import { promisify } from 'node:util';

    const execFileAsync = promisify(execFile);

    interface IndexingState {
      lastIndexedCommit: string;
      collectionName: string;
      indexedAt: string;
    }

    function getStatePath(frameworkDir: string): string {
      return join(frameworkDir, 'indexing-state.json');
    }

    export async function getLastIndexedCommit(frameworkDir: string): Promise<string | null> {
      try {
        const content = await readFile(getStatePath(frameworkDir), 'utf-8');
        const state = JSON.parse(content) as IndexingState;
        return state.lastIndexedCommit;
      } catch {
        return null;
      }
    }

    export async function saveLastIndexedCommit(
      frameworkDir: string,
      commit: string,
      collectionName: string,
    ): Promise<void> {
      await mkdir(frameworkDir, { recursive: true });
      const state: IndexingState = {
        lastIndexedCommit: commit,
        collectionName,
        indexedAt: new Date().toISOString(),
      };
      await writeFile(getStatePath(frameworkDir), JSON.stringify(state, null, 2), 'utf-8');
    }

    export async function getCurrentCommit(projectPath: string): Promise<string | null> {
      try {
        const { stdout } = await execFileAsync('git', ['rev-parse', 'HEAD'], { cwd: projectPath });
        return stdout.trim();
      } catch {
        return null;
      }
    }

    export async function getChangedFiles(
      projectPath: string,
      sinceCommit: string,
    ): Promise<string[]> {
      try {
        const { stdout } = await execFileAsync(
          'git',
          ['diff', '--name-only', `${sinceCommit}...HEAD`],
          { cwd: projectPath },
        );
        return stdout
          .trim()
          .split('\n')
          .filter(Boolean)
          .map((f) => join(projectPath, f));
      } catch {
        return [];
      }
    }
    ```

    ```typescript
    import { parseFile } from '@ia-framework/ast-parser';
    import type { EmbeddingProvider } from '@ia-framework/embedding';
    import type { VectorStore, VectorPoint, ChunkPayload } from '@ia-framework/vector-store';
    import type { CodeChunk } from '@ia-framework/ast-parser';
    import { walkFiles } from './file-walker.js';
    import {
      getLastIndexedCommit,
      saveLastIndexedCommit,
      getCurrentCommit,
      getChangedFiles,
    } from './incremental.js';
    import { createLogger } from '@ia-framework/core';
    import { join } from 'node:path';

    const logger = createLogger('indexer');

    export interface IndexingConfig {
      /** Path to the target project to index */
      targetProjectPath: string;
      /** Path to .ia-framework/ dir for state persistence */
      frameworkDir: string;
      /** Qdrant collection name */
      collectionName: string;
      /** Embedding provider */
      embeddingProvider: EmbeddingProvider;
      /** Vector store */
      vectorStore: VectorStore;
      /** Glob include patterns */
      includePatterns?: string[];
      /** Glob exclude patterns */
      excludePatterns?: string[];
      /** Max chunk lines */
      maxChunkLines?: number;
      /** Embedding batch size */
      embeddingBatchSize?: number;
      /** Use incremental indexing based on git diff */
      incremental?: boolean;
    }

    export interface IndexingProgress {
      stage: 'walking' | 'parsing' | 'embedding' | 'storing' | 'done';
      filesTotal: number;
      filesProcessed: number;
      chunksTotal: number;
      chunksIndexed: number;
      currentFile?: string;
    }

    export type ProgressCallback = (progress: IndexingProgress) => void;

    export class IndexingPipeline {
      async run(
        config: IndexingConfig,
        onProgress?: ProgressCallback,
      ): Promise<{ filesIndexed: number; chunksIndexed: number }> {
        const {
          targetProjectPath,
          frameworkDir,
          collectionName,
          embeddingProvider,
          vectorStore,
          includePatterns,
          excludePatterns,
          maxChunkLines = 200,
          embeddingBatchSize = 50,
          incremental = false,
        } = config;

        // Initialize vector store collection
        await vectorStore.initialize({
          textDimensions: embeddingProvider.dimensions,
          codeDimensions: embeddingProvider.dimensions,
        });

        // Determine which files to process
        let filesToProcess: string[];

        onProgress?.({
          stage: 'walking',
          filesTotal: 0,
          filesProcessed: 0,
          chunksTotal: 0,
          chunksIndexed: 0,
        });

        if (incremental) {
          const lastCommit = await getLastIndexedCommit(frameworkDir);
          if (lastCommit) {
            const changedFiles = await getChangedFiles(targetProjectPath, lastCommit);
            // Delete old chunks for changed files
            for (const filePath of changedFiles) {
              await vectorStore.deleteByFilter({ filePath });
            }
            // Only index changed files that still exist
            filesToProcess = changedFiles;
          } else {
            // No prior state — full index
            filesToProcess = await walkFiles({
              projectPath: targetProjectPath,
              includePatterns,
              excludePatterns,
            });
          }
        } else {
          filesToProcess = await walkFiles({
            projectPath: targetProjectPath,
            includePatterns,
            excludePatterns,
          });
        }

        const progress: IndexingProgress = {
          stage: 'parsing',
          filesTotal: filesToProcess.length,
          filesProcessed: 0,
          chunksTotal: 0,
          chunksIndexed: 0,
        };
        onProgress?.(progress);

        // Parse all files into chunks
        const allChunks: CodeChunk[] = [];
        for (const filePath of filesToProcess) {
          progress.currentFile = filePath;
          try {
            const chunks = await parseFile(filePath, {
              projectRoot: targetProjectPath,
              maxChunkLines,
            });
            allChunks.push(...chunks);
          } catch (err) {
            logger.error({ filePath, err }, 'Failed to parse file — skipping');
          }
          progress.filesProcessed++;
          onProgress?.({ ...progress });
        }

        progress.chunksTotal = allChunks.length;
        progress.stage = 'embedding';
        onProgress?.({ ...progress });

        // Embed chunks in batches
        const vectorPoints: VectorPoint[] = [];
        for (let i = 0; i < allChunks.length; i += embeddingBatchSize) {
          const batch = allChunks.slice(i, i + embeddingBatchSize);
          const texts = batch.map((c) => c.content);

          const vectors = await embeddingProvider.embedBatch(texts);

          for (let j = 0; j < batch.length; j++) {
            const chunk = batch[j]!;
            const vector = vectors[j]!;
            const payload: ChunkPayload = {
              chunkId: chunk.id,
              filePath: chunk.filePath,
              relativePath: chunk.relativePath,
              language: chunk.language,
              chunkType: chunk.chunkType,
              name: chunk.name,
              content: chunk.content,
              startLine: chunk.startLine,
              endLine: chunk.endLine,
              imports: chunk.imports,
              exports: chunk.exports,
              indexedAt: new Date().toISOString(),
            };
            vectorPoints.push({
              id: chunk.id,
              textVector: vector,
              codeVector: vector, // Same model for both in Phase 3
              payload,
            });
          }
          progress.chunksIndexed += batch.length;
          onProgress?.({ ...progress });
        }

        // Store in Qdrant in batches
        progress.stage = 'storing';
        onProgress?.({ ...progress });

        const storeBatchSize = 100;
        for (let i = 0; i < vectorPoints.length; i += storeBatchSize) {
          const batch = vectorPoints.slice(i, i + storeBatchSize);
          await vectorStore.upsertBatch(batch);
        }

        // Save incremental state
        const currentCommit = await getCurrentCommit(targetProjectPath);
        if (currentCommit) {
          await saveLastIndexedCommit(frameworkDir, currentCommit, collectionName);
        }

        progress.stage = 'done';
        progress.currentFile = undefined;
        onProgress?.({ ...progress });

        return {
          filesIndexed: progress.filesProcessed,
          chunksIndexed: vectorPoints.length,
        };
      }
    }
    ```
