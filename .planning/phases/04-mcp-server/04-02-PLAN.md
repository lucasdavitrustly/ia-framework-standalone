<objective>
Create the `packages/mcp-server` package — the MCP server that exposes RAG tools to OpenCode via stdio transport. It registers three tools using `@modelcontextprotocol/sdk`: `search_codebase`, `get_file`, and `get_dependencies`. Each tool delegates to the `SearchEngine` from `@ia-framework/retrieval` and formats results as human-readable code blocks.

Purpose: This is the primary integration point with OpenCode. When the MCP server is running, OpenCode can call these tools to retrieve relevant code context.
Output: `@ia-framework/mcp-server` package with `createMcpServer()` factory and stdio-based MCP server.
</objective>

import { McpServer } from '@modelcontextprotocol/sdk/server/mcp.js';
import { StdioServerTransport } from '@modelcontextprotocol/sdk/server/stdio.js';
import { z } from 'zod';

const server = new McpServer({ name: 'ia-framework', version: '0.0.1' });

server.registerTool('tool_name', {
  description: 'Tool description',
  inputSchema: { query: z.string(), limit: z.number().optional() },
}, async (input) => {
  return {
    content: [{ type: 'text', text: '...' }]
  };
});

const transport = new StdioServerTransport();
await server.connect(transport);
```

<!-- From @ia-framework/retrieval -->
```typescript
class SearchEngine {
  async search(query: RetrievalQuery): Promise<RetrievalResult[]>;
  async getFile(filePath: string): Promise<RetrievalResult[]>;
}

interface RetrievalResult {
  chunkId: string;
  filePath: string;
  relativePath: string;
  language: string;
  chunkType: string;
  name?: string | undefined;
  content: string;
  startLine: number;
  endLine: number;
  imports: string[];
  exports: string[];
  score: number;
  matchedVector: 'text' | 'code' | 'both';
}
```

<!-- Tool output format -->
Each tool returns `{ content: [{ type: 'text', text: string }] }`.
Format results as:
```
## Results for: "authentication middleware"

### 1. src/middleware/auth.ts (authMiddleware) — score: 0.92
```typescript
export function authMiddleware(req, res, next) {
  // ...
}
```

### 2. src/auth/jwt.ts (verifyToken) — score: 0.87
...
```

<!-- Tool schemas -->
search_codebase:
  - query: string (required) — natural language query
  - limit: number (optional, default 10) — max results to return
  - score_threshold: number (optional, default 0.5) — minimum score

get_file:
  - path: string (required) — relative file path (e.g., "src/auth/middleware.ts")

get_dependencies:
  - path: string (required) — relative file path
  Returns: list of imports/exports found in all chunks of that file

    import { registerSearchCodebaseTool } from './tools/search-codebase.js';
    import { registerGetFileTool } from './tools/get-file.js';
    import { registerGetDependenciesTool } from './tools/get-dependencies.js';
    import type { SearchEngine } from '@ia-framework/retrieval';

    export interface McpServerConfig {
      searchEngine: SearchEngine;
      serverName?: string;
      serverVersion?: string;
    }

    export function createMcpServer(config: McpServerConfig): McpServer {
      const server = new McpServer({
        name: config.serverName ?? 'ia-framework',
        version: config.serverVersion ?? '0.0.1',
      });

      registerSearchCodebaseTool(server, config.searchEngine);
      registerGetFileTool(server, config.searchEngine);
      registerGetDependenciesTool(server, config.searchEngine);

      return server;
    }

    #!/usr/bin/env node
    /**
     * MCP server binary — wires config, SearchEngine, and StdioTransport.
     * Run via: node dist/bin.js
     * OpenCode configures this as an MCP server in its config.
     */
    import { StdioServerTransport } from '@modelcontextprotocol/sdk/server/stdio.js';
    import { loadConfig } from '@ia-framework/core';
    import { createEmbeddingProvider } from '@ia-framework/embedding';
    import { QdrantVectorStore } from '@ia-framework/vector-store';
    import { SearchEngine } from '@ia-framework/retrieval';
    import { createMcpServer } from './server.js';
    import { createLogger } from '@ia-framework/core';

    const logger = createLogger('mcp-server');

    async function main() {
      // Load config from .ia-framework/project.json (searches from cwd upward)
      const config = await loadConfig();

      const embeddingProvider = createEmbeddingProvider(config.embedding);
      const vectorStore = new QdrantVectorStore({
        host: config.vectorStore.host,
        port: config.vectorStore.port,
        collectionName: config.vectorStore.collectionName,
        apiKey: config.vectorStore.apiKey,
      });

      const searchEngine = new SearchEngine({ embeddingProvider, vectorStore });
      const server = createMcpServer({ searchEngine });

      const transport = new StdioServerTransport();
      await server.connect(transport);

      logger.info('MCP server started (stdio transport)');
    }

    main().catch((err) => {
      process.stderr.write(`MCP server fatal error: ${err}\n`);
      process.exit(1);
    });
    ```
