    ```typescript
    import type { CrossRepoEdgeStore } from './cross-repo-edges.js';
    import { resolveImportToRepo } from './import-resolver.js';
    import type { RetrievalResult } from '@ia-framework/retrieval';

    export interface EdgeBuilderEngine {
      /** Scroll all chunks for a repo (paginated) */
      scrollAllChunks(repoId: string, batchSize?: number): AsyncIterable<RetrievalResult[]>;
      /** Find chunk by exported symbol in a specific repo */
      scrollByNameInRepo(name: string, repoId: string): Promise<RetrievalResult[]>;
    }

    export interface EdgeBuilderConfig {
      repos: Array<{ name: string; provides?: string[] | undefined }>;
      /** If set, only rebuild edges for this repo (per D-10 — drop+rebuild affected edges) */
      rebuildRepoId?: string;
    }

    /**
     * Second pass: build cross-repo edges after all repos are indexed (per D-09).
     * Iterates chunks in source repo(s), resolves imports against other repos' provides,
     * looks up target symbol, stores edge.
     */
    export async function buildCrossRepoEdges(
      store: CrossRepoEdgeStore,
      engine: EdgeBuilderEngine,
      config: EdgeBuilderConfig,
    ): Promise<{ edgesCreated: number; errors: number }> {
      // Implementation:
      // 1. If config.rebuildRepoId, call store.rebuildForRepo(config.rebuildRepoId)
      // 2. Determine source repos to scan (just rebuildRepoId, or all repos)
      // 3. For each source repo, scroll all chunks
      // 4. For each chunk, for each import in chunk.imports:
      //    a. Call resolveImportToRepo(import, config.repos) excluding current repo
      //    b. If resolution found, call engine.scrollByNameInRepo(symbol, targetRepo)
      //    c. If target found, store.insertEdge({sourceRepo, sourcePath, sourceSymbol, targetRepo, targetPath, targetSymbol})
      // 5. Return stats
    }
    ```

    Key details:
    - Filter out self-repo matches (don't create edges to the same repo)
    - Extract symbol name from import: last segment of remaining path, or the specifier itself
    - Batch inserts using store.insertEdges for performance
    - Silently skip unresolved imports (no error, just increment counter)

    Export from `packages/core/src/index.ts`: `export { buildCrossRepoEdges, type EdgeBuilderEngine, type EdgeBuilderConfig } from './graph/cross-repo-edge-builder.js';`
