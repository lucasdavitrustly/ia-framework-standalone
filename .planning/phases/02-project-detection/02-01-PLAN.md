Detection strategy:
- **Package manager**: check for pnpm-lock.yaml (pnpm), yarn.lock (yarn), bun.lockb (bun), package-lock.json (npm), requirements.txt/pyproject.toml (pip/poetry/uv), go.sum (go mod)
- **Languages**: scan file extensions in root + src/ dirs (avoid node_modules, .git): .ts/.tsx → typescript, .js/.jsx → javascript, .py → python, .go → go, .rs → rust, .java → java
- **Frameworks**: read package.json dependencies + devDependencies (if exists): 'next' → next.js, 'react' → react, 'vue' → vue, 'express' → express, 'fastify' → fastify, 'hono' → hono, 'nest' → nestjs, '@angular/core' → angular. For Python: check pyproject.toml for django, fastapi, flask.
- **Test runner**: devDependencies: 'vitest' → vitest, 'jest' → jest, 'mocha' → mocha, '@playwright/test' → playwright, 'pytest' → pytest (requirements.txt), 'go test' → go (always)
- **Build command**: look for 'build' key in package.json scripts → `pnpm build` / `npm run build`. For Python: check pyproject.toml. For Go: `go build`.
- **Dev command**: look for 'dev' or 'start' key in package.json scripts.

    import { access } from 'node:fs/promises';
    import { join } from 'node:path';

    const LOCKFILES: Array<{ file: string; manager: string }> = [
      { file: 'pnpm-lock.yaml', manager: 'pnpm' },
      { file: 'yarn.lock', manager: 'yarn' },
      { file: 'bun.lockb', manager: 'bun' },
      { file: 'package-lock.json', manager: 'npm' },
      { file: 'go.sum', manager: 'go mod' },
      { file: 'Pipfile.lock', manager: 'pipenv' },
      { file: 'poetry.lock', manager: 'poetry' },
      { file: 'uv.lock', manager: 'uv' },
    ];

    async function exists(filePath: string): Promise<boolean> {
      try {
        await access(filePath);
        return true;
      } catch {
        return false;
      }
    }

    export async function detectPackageManager(projectPath: string): Promise<string> {
      for (const { file, manager } of LOCKFILES) {
        if (await exists(join(projectPath, file))) {
          return manager;
        }
      }
      // Fallback: check for package.json without lockfile
      if (await exists(join(projectPath, 'package.json'))) {
        return 'npm';
      }
      return 'unknown';
    }

    **packages/project-detector/src/detectors/language.ts**:
    ```typescript
    import { readdir, stat } from 'node:fs/promises';
    import { join, extname } from 'node:path';

    const EXTENSION_MAP: Record<string, string> = {
      '.ts': 'typescript',
      '.tsx': 'typescript',
      '.js': 'javascript',
      '.jsx': 'javascript',
      '.mjs': 'javascript',
      '.cjs': 'javascript',
      '.py': 'python',
      '.go': 'go',
      '.rs': 'rust',
      '.java': 'java',
      '.kt': 'kotlin',
      '.rb': 'ruby',
      '.php': 'php',
      '.cs': 'csharp',
      '.cpp': 'cpp',
      '.c': 'c',
      '.swift': 'swift',
    };

    const SKIP_DIRS = new Set([
      'node_modules', '.git', 'dist', 'build', '.next', 'out',
      '__pycache__', '.venv', 'venv', 'target', '.turbo', 'coverage',
    ]);

    async function scanDir(dirPath: string, counts: Map<string, number>, depth = 0): Promise<void> {
      if (depth > 3) return; // limit scan depth
      let entries: Awaited<ReturnType<typeof readdir>>;
      try {
        entries = await readdir(dirPath, { withFileTypes: true });
      } catch {
        return;
      }
      for (const entry of entries) {
        if (entry.isDirectory()) {
          if (!SKIP_DIRS.has(entry.name) && !entry.name.startsWith('.')) {
            await scanDir(join(dirPath, entry.name), counts, depth + 1);
          }
        } else if (entry.isFile()) {
          const ext = extname(entry.name).toLowerCase();
          const lang = EXTENSION_MAP[ext];
          if (lang) {
            counts.set(lang, (counts.get(lang) ?? 0) + 1);
          }
        }
      }
    }

    export async function detectLanguages(projectPath: string): Promise<string[]> {
      const counts = new Map<string, number>();
      await scanDir(projectPath, counts);
      // Sort by file count descending, return languages with >0 files
      return [...counts.entries()]
        .sort((a, b) => b[1] - a[1])
        .filter(([, count]) => count > 0)
        .map(([lang]) => lang);
    }

      { dep: 'next', framework: 'next.js', exact: true },
      { dep: 'nuxt', framework: 'nuxt', exact: true },
      { dep: '@nuxt/core', framework: 'nuxt' },
      { dep: 'react', framework: 'react', exact: true },
      { dep: 'vue', framework: 'vue', exact: true },
      { dep: 'svelte', framework: 'svelte', exact: true },
      { dep: '@angular/core', framework: 'angular' },
      { dep: 'express', framework: 'express', exact: true },
      { dep: 'fastify', framework: 'fastify', exact: true },
      { dep: 'hono', framework: 'hono', exact: true },
      { dep: '@nestjs/core', framework: 'nestjs' },
      { dep: 'koa', framework: 'koa', exact: true },
      { dep: 'remix', framework: 'remix', exact: true },
      { dep: '@remix-run/node', framework: 'remix' },
      { dep: 'astro', framework: 'astro', exact: true },
      { dep: 'vite', framework: 'vite', exact: true },
      { dep: 'electron', framework: 'electron', exact: true },
    ];

    export async function detectFrameworks(projectPath: string): Promise<string[]> {
      let pkgJson: PackageJson;
      try {
        const raw = await readFile(join(projectPath, 'package.json'), 'utf-8');
        pkgJson = JSON.parse(raw) as PackageJson;
      } catch {
        return [];
      }

      const allDeps = {
        ...pkgJson.dependencies,
        ...pkgJson.devDependencies,
      };

      const frameworks = new Set<string>();
      for (const { dep, framework } of FRAMEWORK_MAP) {
        if (dep in allDeps) {
          frameworks.add(framework);
        }
      }
      return [...frameworks];
    }

      dependencies?: Record<string, string>;
      devDependencies?: Record<string, string>;
    }

    const TEST_RUNNER_DEPS: Array<{ dep: string; runner: string }> = [
      { dep: 'vitest', runner: 'vitest' },
      { dep: 'jest', runner: 'jest' },
      { dep: '@jest/core', runner: 'jest' },
      { dep: 'mocha', runner: 'mocha' },
      { dep: '@playwright/test', runner: 'playwright' },
      { dep: 'cypress', runner: 'cypress' },
      { dep: 'jasmine', runner: 'jasmine' },
      { dep: 'ava', runner: 'ava' },
      { dep: 'tap', runner: 'tap' },
    ];

    async function exists(filePath: string): Promise<boolean> {
      try { await access(filePath); return true; } catch { return false; }
    }

    export async function detectTestRunner(projectPath: string): Promise<string | undefined> {
      // Check package.json deps first
      try {
        const raw = await readFile(join(projectPath, 'package.json'), 'utf-8');
        const pkgJson = JSON.parse(raw) as PackageJson;
        const allDeps = { ...pkgJson.dependencies, ...pkgJson.devDependencies };
        for (const { dep, runner } of TEST_RUNNER_DEPS) {
          if (dep in allDeps) return runner;
        }
      } catch {
        // No package.json — check other ecosystems
      }

      // Python
      if (await exists(join(projectPath, 'pytest.ini')) ||
          await exists(join(projectPath, 'pyproject.toml'))) {
        return 'pytest';
      }

      // Go always has go test
      if (await exists(join(projectPath, 'go.mod'))) {
        return 'go test';
      }

      return undefined;
    }

    import { readFile } from 'node:fs/promises';
    import { join } from 'node:path';
    import { detectPackageManager } from './package-manager.js';

    interface PackageJson {
      scripts?: Record<string, string>;
    }

    export async function detectCommands(projectPath: string): Promise<{
      buildCommand: string | undefined;
      devCommand: string | undefined;
    }> {
      const pm = await detectPackageManager(projectPath);
      const runPrefix = pm === 'npm' ? 'npm run' : pm === 'bun' ? 'bun run' : `${pm}`;

      let scripts: Record<string, string> = {};
      try {
        const raw = await readFile(join(projectPath, 'package.json'), 'utf-8');
        const pkgJson = JSON.parse(raw) as PackageJson;
        scripts = pkgJson.scripts ?? {};
      } catch {
        // No package.json
      }

      const buildCommand = 'build' in scripts ? `${runPrefix} build` : undefined;
      const devCommand = 'dev' in scripts
        ? `${runPrefix} dev`
        : 'start' in scripts
        ? `${runPrefix} start`
        : undefined;

      return { buildCommand, devCommand };
    }

    import { access } from 'node:fs/promises';
    import type { ProjectProfile } from '@ia-framework/core';
    import { detectLanguages } from './detectors/language.js';
    import { detectPackageManager } from './detectors/package-manager.js';
    import { detectFrameworks } from './detectors/framework.js';
    import { detectTestRunner } from './detectors/test-runner.js';
    import { detectCommands } from './detectors/commands.js';

    export class DetectionError extends Error {
      constructor(
        public readonly projectPath: string,
        cause: unknown,
      ) {
        super(`Cannot detect project at path: ${projectPath}`);
        this.name = 'DetectionError';
        this.cause = cause;
      }
    }

    export interface DetectionResult extends ProjectProfile {
      /** Absolute path that was scanned */
      scannedPath: string;
    }

    export async function detectProject(projectPath: string): Promise<DetectionResult> {
      // Verify path exists
      try {
        await access(projectPath);
      } catch (err) {
        throw new DetectionError(projectPath, err);
      }

      const [languages, packageManager, frameworks, testRunner, { buildCommand, devCommand }] =
        await Promise.all([
          detectLanguages(projectPath),
          detectPackageManager(projectPath),
          detectFrameworks(projectPath),
          detectTestRunner(projectPath),
          detectCommands(projectPath),
        ]);

      return {
        scannedPath: projectPath,
        languages,
        packageManager,
        frameworks,
        testRunner,
        buildCommand,
        devCommand,
      };
    }
    ```
