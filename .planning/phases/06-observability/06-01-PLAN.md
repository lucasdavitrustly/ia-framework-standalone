<!-- TokenCounter — simple char-based heuristic -->
```typescript
// ~4 characters per token is a reasonable approximation for English text
// and code mixed content. Good enough for cost-visibility estimates.
export class TokenCounter {
  private readonly charsPerToken: number;

  constructor(charsPerToken = 4) {
    this.charsPerToken = charsPerToken;
  }

  /** Estimate token count for a single text string */
  count(text: string): number {
    return Math.ceil(text.length / this.charsPerToken);
  }

  /** Estimate total tokens for a query + array of chunk content strings */
  countRagEvent(query: string, chunkContents: string[]): number {
    const total = query + chunkContents.join('');
    return this.count(total);
  }
}
```

<!-- better-sqlite3 API notes -->
- Install: `better-sqlite3` + `@types/better-sqlite3`
- Import: `import Database from 'better-sqlite3'`
- Synchronous: `db.prepare(sql).run(params)` and `db.prepare(sql).get(params)`
- For queries returning multiple rows: `db.prepare(sql).all(params)`
- All operations are synchronous — no async/await needed
- Use `JSON.stringify(arr)` / `JSON.parse(str)` for chunk_ids TEXT column
