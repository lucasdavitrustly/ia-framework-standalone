<!-- Session ID generation -->
```typescript
// packages/observability/src/session.ts
// Session ID = ISO date (YYYY-MM-DD) + '-' + process.pid
// e.g. "2026-04-01-12345"
// This ensures:
//   - Unique per process (pid)
//   - Human-readable (date portion)
//   - Stable for the entire duration of the process
export function generateSessionId(): string {
  const date = new Date().toISOString().slice(0, 10); // YYYY-MM-DD
  return `${date}-${process.pid}`;
}
```

<!-- MCP server search tool logging pattern -->
```typescript
// In the search_codebase tool handler:
const startTime = Date.now();
const results = await searchEngine.search(query);
const latencyMs = Date.now() - startTime;

const tokenCount = tokenCounter.countRagEvent(
  queryText,
  results.map(r => r.content)
);

metricsStore.logEvent({
  id: crypto.randomUUID(),
  sessionId,
  query: queryText,
  chunksRetrieved: results.length,
  chunkIds: results.map(r => r.chunkId),
  tokenCount,
  strategy: 'mcp',
  timestamp: new Date().toISOString(),
  latencyMs,
});
```
