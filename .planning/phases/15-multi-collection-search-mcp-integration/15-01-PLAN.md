    ```typescript
    import type { RetrievalQuery, RetrievalResult } from './types.js';
    import type { SearchEngine } from './search-engine.js';
    import { rrfMerge } from './rrf.js';

    export interface CollectionEngine {
      name: 'ast' | 'llm';
      engine: SearchEngine;
    }

    export interface MultiCollectionSearchEngineConfig {
      engines: CollectionEngine[];
      dependencyGraph?: {
        getNeighborFiles(filePaths: string[], hops: 1 | 2): Promise<string[]>;
      };
    }

    export class MultiCollectionSearchEngine {
      private readonly engines: CollectionEngine[];
      private readonly dependencyGraph?: { getNeighborFiles(filePaths: string[], hops: 1 | 2): Promise<string[]> };

      constructor(config: MultiCollectionSearchEngineConfig) { ... }
      async search(query: RetrievalQuery): Promise<RetrievalResult[]> { ... }
      async searchHybrid(query: RetrievalQuery): Promise<RetrievalResult[]> { ... }
      async searchWithExpansion(query: RetrievalQuery, options?: { hops?: 0 | 1 | 2 }): Promise<RetrievalResult[]> { ... }
      async getFile(relativePath: string): Promise<RetrievalResult[]> { ... }
    }
    ```

    **search() implementation:**
    - Over-fetch: pass `{ ...query, limit: (query.limit ?? 10) * 2 }` to each engine.search()
    - Run all engines in parallel via `Promise.all`
    - Map results to RankedList[] for rrfMerge
    - Call `rrfMerge(lists, query.limit ?? 10)`
    - Return merged results

    **searchHybrid() implementation:**
    - Same pattern as search() but calls engine.searchHybrid()
    - Over-fetch: pass `{ ...query, limit: (query.limit ?? 10) * 2 }` to each engine

    **searchWithExpansion() implementation:**
    - Call `this.search(query)` for base multi-collection results
    - If no dependencyGraph or hops === 0, return base results
    - Collect unique file paths from base results
    - Get neighbor files via dependencyGraph.getNeighborFiles()
    - For expansion: call `this.getFile()` for each new file (or use a similar pattern to SearchEngine.fetchChunksForFiles — but since we don't have direct vectorStore access, use getFile for each neighbor file path)
    - Actually, better approach: call the FIRST engine's underlying searchWithExpansion directly? No — that would only expand from one collection.
    - Correct approach: Run `this.search(query)` first (multi-collection). Then for expansion, call getFile on each neighbor file through each engine, combine and dedup. But this is complex.
    - Simplest correct approach per CONTEXT.md D-03: "Run the base multi-collection search, then expand from the merged results' file paths using the single shared dependency graph." The expansion fetches chunks via getFile (which queries all collections). So:
      1. `baseResults = await this.search(query)`
      2. If dependencyGraph and hops > 0: get neighbor files
      3. For each neighbor file not in base results: `await this.getFile(filePath)` → tag each with score=0
      4. Return [...baseResults, ...expansionResults]

    **getFile() implementation:**
    - Call engine.getFile(relativePath) on all engines in parallel
    - Tag each result with source from the engine's name
    - Combine all results, dedup by chunkId (keep first)
    - Sort by startLine for natural reading order
