1. **Java: annotated method includes annotations**

   ```typescript
   it('includes @Transactional and @CacheEvict in Java method chunk', async () => {
     const filePath = await makeFile(
       'Service.java',
       `
   package com.example;
   
   public class OrderService {
       @Transactional
       @CacheEvict(value = "orders")
       public void updateOrder(Long id, OrderDto dto) {
           Order order = repo.findById(id).orElseThrow();
           order.update(dto);
           repo.save(order);
       }
   }
     `.trim(),
     );
     const chunks = await parseFile(filePath, { projectRoot: tmpDir });
     const methodChunk = chunks.find((c) => c.name === 'updateOrder');
     expect(methodChunk).toBeDefined();
     expect(methodChunk!.content).toContain('@Transactional');
     expect(methodChunk!.content).toContain('@CacheEvict');
   });
   ```

2. **Java: @Aspect class → single 'aspect' chunk**

   ```typescript
   it('produces single aspect chunk for @Aspect class', async () => {
     const filePath = await makeFile(
       'AuditAspect.java',
       `
   package com.example;
   
   import org.aspectj.lang.annotation.Aspect;
   import org.aspectj.lang.annotation.Around;
   import org.aspectj.lang.ProceedingJoinPoint;
   import org.springframework.stereotype.Component;
   
   @Aspect
   @Component
   public class AuditAspect {
       @Around("execution(* com.example.service.*.*(..))")
       public Object audit(ProceedingJoinPoint jp) throws Throwable {
           return jp.proceed();
       }
   
       @Before("@annotation(Auditable)")
       public void beforeAudit(JoinPoint jp) {
           System.out.println(jp.getSignature());
       }
   }
     `.trim(),
     );
     const chunks = await parseFile(filePath, { projectRoot: tmpDir });
     const aspectChunks = chunks.filter((c) => c.chunkType === 'aspect');
     expect(aspectChunks).toHaveLength(1);
     expect(aspectChunks[0].content).toContain('@Aspect');
     expect(aspectChunks[0].content).toContain('@Around');
     expect(aspectChunks[0].content).toContain('AuditAspect');
   });
   ```

3. **Java: regular class still splits methods (not treated as aspect)**

   ```typescript
   it('does not treat regular class as aspect', async () => {

   });
   ```
