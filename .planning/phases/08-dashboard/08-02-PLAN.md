    ### DashboardClient object

    ```typescript
    const BASE = '';  // Same origin via Vite proxy (or relative path in production)

    async function get<T>(path: string): Promise<T> {
      const res = await fetch(`${BASE}${path}`);
      if (!res.ok) throw new Error(`API error ${res.status}: ${path}`);
      return res.json() as Promise<T>;
    }

    async function post<T>(path: string, body?: unknown): Promise<T> {
      const res = await fetch(`${BASE}${path}`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: body !== undefined ? JSON.stringify(body) : undefined,
      });
      if (!res.ok) {
        const err = await res.json().catch(() => ({})) as { error?: string };
        throw new Error(err.error ?? `API error ${res.status}: ${path}`);
      }
      return res.json() as Promise<T>;
    }

    export const DashboardClient = {
      getStatus: () => get<ApiStatus>('/api/status'),
      getConfig: () => get<ProjectConfig>('/api/config'),
      getMetrics: () => get<ApiMetrics>('/api/metrics'),
      getMetricsEvents: (params?: { limit?: number; offset?: number }) => {
        const qs = new URLSearchParams();
        if (params?.limit !== undefined) qs.set('limit', String(params.limit));
        if (params?.offset !== undefined) qs.set('offset', String(params.offset));
        const q = qs.toString();
        return get<ApiMetricsEvents>(`/api/metrics/events${q ? `?${q}` : ''}`);
      },
      getBenchmarks: (limit?: number) => get<ApiBenchmarks>(`/api/benchmarks${limit ? `?limit=${limit}` : ''}`),
      getBenchmarkRun: (runId: string) => get<BenchmarkRun>(`/api/benchmarks/${runId}`),
      getIndexingStatus: () => get<ApiIndexingStatus>('/api/indexing/status'),
      triggerIndexing: (body?: { incremental?: boolean; label?: string }) =>
        post<ApiTriggerResponse>('/api/indexing/trigger', body),
    };
    ```

    Fetches status + metrics + benchmarks on mount. Shows 4 stat cards in a grid.

    ```tsx
    import { useEffect, useState } from 'react';
    import StatCard from '../components/StatCard.tsx';
    import { DashboardClient, ApiStatus, ApiMetrics, ApiBenchmarks } from '../api/client.ts';

    export default function Overview() {
      const [status, setStatus] = useState<ApiStatus | null>(null);
      const [metrics, setMetrics] = useState<ApiMetrics | null>(null);
      const [benchmarks, setBenchmarks] = useState<ApiBenchmarks | null>(null);
      const [loading, setLoading] = useState(true);
      const [error, setError] = useState(false);

      useEffect(() => {
        Promise.all([
          DashboardClient.getStatus(),
          DashboardClient.getMetrics(),
          DashboardClient.getBenchmarks(1),
        ])
          .then(([s, m, b]) => { setStatus(s); setMetrics(m); setBenchmarks(b); })
          .catch(() => setError(true))
          .finally(() => setLoading(false));
      }, []);

      const lastMrr = benchmarks?.runs[0]?.mrr;
      const qdrantOk = status?.qdrant === 'ok';

      return (
        <div className="p-8">
          <h2 className="text-xl font-semibold text-gray-100 mb-6">Overview</h2>

          {/* Qdrant health banner */}
          {!loading && !qdrantOk && (
            <div className="mb-4 p-3 bg-red-900/40 border border-red-700 rounded-lg text-sm text-red-300">
              Qdrant is unreachable — check that Docker is running
            </div>
          )}

          <div className="grid grid-cols-2 gap-4 lg:grid-cols-4">
            <StatCard
              label="Total Tokens"
              value={metrics ? metrics.totals.totalTokens.toLocaleString() : 0}
              sub="all sessions"
              loading={loading}
              error={error}
            />
            <StatCard
              label="Total Queries"
              value={metrics ? metrics.totals.totalEvents : 0}
              sub={`${metrics?.totals.totalSessions ?? 0} sessions`}
              loading={loading}
              error={error}
            />
            <StatCard
              label="Avg Tokens/Query"
              value={metrics ? metrics.totals.avgTokensPerQuery : 0}
              loading={loading}
              error={error}
            />
            <StatCard
              label="Best Benchmark MRR"
              value={lastMrr !== undefined ? lastMrr.toFixed(3) : '—'}
              sub={benchmarks?.runs[0]?.label ?? ''}
              loading={loading}
              error={error}
            />
          </div>

          {/* Indexing status indicator */}
          {status && (
            <div className="mt-8 p-4 bg-gray-900 rounded-xl border border-gray-800">
              <p className="text-xs font-medium text-gray-500 uppercase tracking-wider mb-2">Indexing</p>
              <div className="flex items-center gap-2">
                <span className={`inline-block w-2 h-2 rounded-full ${status.indexingInProgress ? 'bg-yellow-400 animate-pulse' : 'bg-green-500'}`} />
                <span className="text-sm text-gray-300">{status.indexingInProgress ? 'In progress' : 'Idle'}</span>

                )}
              </div>
            </div>
          )}
        </div>
      );
    }
    ```

    ### src/pages/Config.tsx (DASH-05 — fully implemented)

    Fetches GET /api/config and displays the ProjectProfile + embedding config:

    ```tsx
    import { useEffect, useState } from 'react';
    import { DashboardClient, ProjectConfig } from '../api/client.ts';

    function Field({ label, value }: { label: string; value: string | undefined | null }) {
      return (
        <div className="py-3 border-b border-gray-800 last:border-0">
          <p className="text-xs text-gray-500 uppercase tracking-wider">{label}</p>
          <p className="mt-0.5 text-sm text-gray-200">{value ?? <span className="text-gray-600">—</span>}</p>
        </div>
      );
    }

    export default function Config() {
      const [config, setConfig] = useState<ProjectConfig | null>(null);
      const [loading, setLoading] = useState(true);
      const [error, setError] = useState<string | null>(null);

      useEffect(() => {
        DashboardClient.getConfig()
          .then(setConfig)
          .catch((e: Error) => setError(e.message))
          .finally(() => setLoading(false));
      }, []);

      if (loading) return <div className="p-8 text-gray-500 text-sm">Loading config…</div>;
      if (error) return <div className="p-8 text-red-400 text-sm">Failed to load config: {error}</div>;
      if (!config) return null;

      const profile = config.profile;

      return (
        <div className="p-8 max-w-2xl">
          <h2 className="text-xl font-semibold text-gray-100 mb-6">Project Config</h2>

          {/* Project profile */}
          <section className="bg-gray-900 rounded-xl border border-gray-800 p-5 mb-6">
            <h3 className="text-xs font-semibold text-indigo-400 uppercase tracking-wider mb-3">Project Profile</h3>
            <Field label="Target Project" value={config.targetProjectPath} />
            <Field label="Languages" value={profile?.languages.join(', ')} />
            <Field label="Frameworks" value={profile?.frameworks.join(', ')} />
            <Field label="Package Manager" value={profile?.packageManager} />
            <Field label="Test Runner" value={profile?.testRunner} />
            <Field label="Build Command" value={profile?.buildCommand} />
            <Field label="Dev Command" value={profile?.devCommand} />
          </section>

          {/* Embedding config */}
          <section className="bg-gray-900 rounded-xl border border-gray-800 p-5 mb-6">
            <h3 className="text-xs font-semibold text-indigo-400 uppercase tracking-wider mb-3">Embedding</h3>
            <Field label="Provider" value={config.embedding.provider} />
            <Field label="Model" value={config.embedding.model} />
          </section>

          {/* Vector store config */}
          <section className="bg-gray-900 rounded-xl border border-gray-800 p-5">
            <h3 className="text-xs font-semibold text-indigo-400 uppercase tracking-wider mb-3">Vector Store (Qdrant)</h3>
            <Field label="Host" value={config.vectorStore.host} />
            <Field label="Port" value={String(config.vectorStore.port)} />
            <Field label="Collection" value={config.vectorStore.collectionName} />
          </section>
        </div>
      );
    }
    ```
