    ### Updated docker-compose.yml

    Add `dashboard` service and a `.env` variable note for `IA_FRAMEWORK_DIR`. Also add the `dashboard` service to the existing file:

    ```yaml
    services:
      qdrant:
        image: qdrant/qdrant:latest
        container_name: ia-framework-qdrant
        restart: unless-stopped
        ports:
          - "${QDRANT_HTTP_PORT:-6333}:6333"
          - "${QDRANT_GRPC_PORT:-6334}:6334"
        volumes:
          - qdrant_data:/qdrant/storage
        environment:
          QDRANT__SERVICE__GRPC_PORT: "6334"
          QDRANT__LOG_LEVEL: INFO
        ulimits:
          nofile:
            soft: 65536
            hard: 65536
        healthcheck:
          test: ["CMD-SHELL", "curl -f http://localhost:6333/readyz || exit 1"]
          interval: 30s
          timeout: 10s
          retries: 5
          start_period: 10s

      dashboard:
        build:
          context: .
          dockerfile: packages/dashboard-api/Dockerfile
        image: ia-framework-dashboard:latest
        container_name: ia-framework-dashboard
        restart: unless-stopped
        ports:
          - "${DASHBOARD_PORT:-3000}:3000"
        volumes:
          # Bind-mount the target project's .ia-framework/ directory into the container
          # IA_FRAMEWORK_DIR must be set in the shell environment or .env file
          - "${IA_FRAMEWORK_DIR}:/ia-framework-data"
        environment:
          DASHBOARD_PORT: "3000"
          IA_FRAMEWORK_DIR: "/ia-framework-data"
          # Qdrant is accessible via Docker network (service name resolves inside compose)
          # project.json sets vectorStore.host — override if needed:
          # QDRANT_HOST: "qdrant"
        depends_on:
          qdrant:
            condition: service_healthy
        healthcheck:
          test: ["CMD", "node", "-e", "fetch('http://localhost:3000/api/status').then(r => r.ok ? process.exit(0) : process.exit(1)).catch(() => process.exit(1))"]
          interval: 30s
          timeout: 10s
          retries: 3
          start_period: 15s

    volumes:
      qdrant_data:
        name: ia-framework-qdrant-data
    ```

    **Key design decisions:**
    - `IA_FRAMEWORK_DIR` on host → `/ia-framework-data` in container
    - Container sets `IA_FRAMEWORK_DIR=/ia-framework-data` so `server.ts` reads SQLite from the mounted path
    - `depends_on: qdrant: condition: service_healthy` ensures Qdrant is ready before dashboard starts
    - `DASHBOARD_PORT` is configurable via host env var (default 3000)
    - Qdrant host: dashboard container needs to reach Qdrant via the Docker internal network. The `project.json` `vectorStore.host` may say `localhost` (set during install for CLI use). For Docker-internal access, the hostname should be `qdrant` (service name). Add a comment explaining developers may need to update `project.json` `vectorStore.host` to `qdrant` if they want the dashboard container's re-index feature to reach Qdrant.

    Create `.env.example` at repo root for documentation:
    ```
    # ia-framework environment variables
    # Copy to .env and fill in values

    # Required: Path to the target project's .ia-framework/ directory
    IA_FRAMEWORK_DIR=/path/to/your-project/.ia-framework

    # Optional: Dashboard port (default: 3000)
    DASHBOARD_PORT=3000

    # Optional: Qdrant ports (defaults: 6333, 6334)
    QDRANT_HTTP_PORT=6333
    QDRANT_GRPC_PORT=6334
    ```

    local retries=0
