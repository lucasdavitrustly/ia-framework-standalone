    Props: `{ data: DailyTokenUsage[] }`

    - If `data.length === 0`: render a centered message `"No token usage recorded yet"` inside the chart container
    - Dark theme: background transparent, grid lines `#374151`, axis ticks `#9ca3af`, bar fill `#6366f1`
    - Tooltip shows: date, totalTokens (formatted with toLocaleString()), totalQueries
    - Custom tooltip formatter: label the values as "Tokens" and "Queries"
    - X-axis: show date as MM-DD (slice last 5 chars of YYYY-MM-DD)
    - Y-axis: format large numbers as "12.5k" for values ≥ 1000

    ```tsx
    import { BarChart, Bar, XAxis, YAxis, CartesianGrid, Tooltip, ResponsiveContainer } from 'recharts';
    import type { DailyTokenUsage } from '../api/client.ts';

    interface Props { data: DailyTokenUsage[]; }

    function formatY(value: number): string {
      return value >= 1000 ? `${(value / 1000).toFixed(1)}k` : String(value);
    }

    export default function DailyTokenChart({ data }: Props) {
      if (data.length === 0) {
        return (
          <div className="h-72 flex items-center justify-center text-gray-600 text-sm">
            No token usage recorded yet
          </div>
        );
      }

      const chartData = data.map(d => ({ ...d, date: d.date.slice(5) })); // MM-DD

      return (
        <ResponsiveContainer width="100%" height={288}>
          <BarChart data={chartData} margin={{ top: 4, right: 16, bottom: 0, left: 8 }}>
            <CartesianGrid strokeDasharray="3 3" stroke="#374151" vertical={false} />
            <XAxis dataKey="date" tick={{ fill: '#9ca3af', fontSize: 12 }} axisLine={false} tickLine={false} />
            <YAxis tickFormatter={formatY} tick={{ fill: '#9ca3af', fontSize: 12 }} axisLine={false} tickLine={false} />
            <Tooltip
              contentStyle={{ backgroundColor: '#111827', border: '1px solid #374151', borderRadius: '8px' }}
              labelStyle={{ color: '#e5e7eb' }}
              formatter={(value: number, name: string) => [
                value.toLocaleString(),
                name === 'totalTokens' ? 'Tokens' : 'Queries',
              ]}
            />
            <Bar dataKey="totalTokens" fill="#6366f1" radius={[4, 4, 0, 0]} name="totalTokens" />
          </BarChart>
        </ResponsiveContainer>
      );
    }
    ```

    ### QueryEventTable.tsx

    Props: `{ initialLimit?: number }` — fetches its own data via DashboardClient

    State:
    - `events: RagEvent[]`
    - `total: number`
    - `offset: number` (pagination)
    - `limit: number` (default 20)
    - `expanded: string | null` (event ID of expanded row)
    - `loading: boolean`

    Behavior:
    - On mount and whenever `offset` changes: calls `DashboardClient.getMetricsEvents({ limit, offset })`
    - Table columns: Timestamp (format as HH:MM:SS of the ISO string), Query (truncated to 40 chars), Strategy, Tokens, Chunks, Latency
    - Clicking a row expands it in-place, showing full query text + all chunkIds
    - Pagination: "← Prev" / "Next →" buttons below table, disabled when at boundary
    - Empty state: "No queries recorded yet"
    - Timestamps formatted as local time (new Date(ts).toLocaleTimeString())

    ```tsx
    import { useEffect, useState } from 'react';
    import { DashboardClient, RagEvent } from '../api/client.ts';

    interface Props { initialLimit?: number; }

    export default function QueryEventTable({ initialLimit = 20 }: Props) {
      const [events, setEvents] = useState<RagEvent[]>([]);
      const [total, setTotal] = useState(0);
      const [offset, setOffset] = useState(0);
      const [expanded, setExpanded] = useState<string | null>(null);
      const [loading, setLoading] = useState(true);
      const limit = initialLimit;

      useEffect(() => {
        setLoading(true);
        DashboardClient.getMetricsEvents({ limit, offset })
          .then(res => { setEvents(res.events); setTotal(res.total); })
          .catch(() => {})
          .finally(() => setLoading(false));
      }, [offset, limit]);

      if (loading) return <div className="h-24 bg-gray-800/50 rounded-lg animate-pulse" />;

      if (events.length === 0) {
        return <p className="text-sm text-gray-600 py-4">No queries recorded yet</p>;
      }

      return (
        <div>
          <div className="overflow-x-auto rounded-lg border border-gray-800">
            <table className="w-full text-sm text-gray-300">
              <thead className="bg-gray-800/60 text-xs text-gray-500 uppercase tracking-wider">
                <tr>
                  {['Time', 'Query', 'Strategy', 'Tokens', 'Chunks', 'Latency'].map(h => (
                    <th key={h} className="px-4 py-3 text-left">{h}</th>
                  ))}
                </tr>
              </thead>
              <tbody>
                {events.map(e => (
                  <>
                    <tr
                      key={e.id}
                      onClick={() => setExpanded(expanded === e.id ? null : e.id)}
                      className="border-t border-gray-800 hover:bg-gray-800/40 cursor-pointer"
                    >
                      <td className="px-4 py-2 whitespace-nowrap text-gray-500">{new Date(e.timestamp).toLocaleTimeString()}</td>
                      <td className="px-4 py-2 max-w-xs truncate">{e.query}</td>
                      <td className="px-4 py-2"><span className="px-2 py-0.5 rounded text-xs bg-indigo-900/50 text-indigo-300">{e.strategy}</span></td>
                      <td className="px-4 py-2">{e.tokenCount.toLocaleString()}</td>
                      <td className="px-4 py-2">{e.chunksRetrieved}</td>
                      <td className="px-4 py-2">{e.latencyMs}ms</td>
                    </tr>
                    {expanded === e.id && (
                      <tr key={`${e.id}-expanded`} className="bg-gray-900">
                        <td colSpan={6} className="px-4 py-3 text-xs text-gray-400 border-t border-gray-800">
                          <p className="mb-1"><span className="text-gray-500">Full query:</span> {e.query}</p>
                          <p><span className="text-gray-500">Chunk IDs:</span> {e.chunkIds.join(', ') || '—'}</p>
                        </td>
                      </tr>
                    )}
                  </>
                ))}
              </tbody>
            </table>
          </div>

          {/* Pagination */}
          <div className="flex items-center justify-between mt-3 text-xs text-gray-500">
            <span>{offset + 1}–{Math.min(offset + limit, total)} of {total}</span>
            <div className="flex gap-2">
              <button
                disabled={offset === 0}
                onClick={() => setOffset(Math.max(0, offset - limit))}
                className="px-3 py-1 rounded bg-gray-800 disabled:opacity-40 hover:bg-gray-700"
              >
                ← Prev
              </button>
              <button
                disabled={offset + limit >= total}
                onClick={() => setOffset(offset + limit)}
                className="px-3 py-1 rounded bg-gray-800 disabled:opacity-40 hover:bg-gray-700"
              >
                Next →
              </button>
            </div>
          </div>
        </div>
      );
    }
    ```

    ### IndexingPanel.tsx

    Props: none — fetches and polls its own data.

    State:
    - `status: ApiIndexingStatus | null`
    - `loading: boolean`
    - `triggering: boolean` (POST in flight)
    - `triggerError: string | null`

    Polling logic:
    - On mount: call `DashboardClient.getIndexingStatus()` once
    - While `status.running === true`: poll every 2000ms via `setInterval`
    - When status transitions from `running=true` to `running=false`: stop polling
    - Clear interval on component unmount (`useEffect` cleanup)
    - Use `useRef` to hold the interval ID

    Trigger button logic:
    - Label: `"Trigger Re-index"` when idle, `"Indexing…"` when triggering or running
    - Disabled when `status.running === true` or `triggering === true`
    - On click: set `triggering=true`, call `DashboardClient.triggerIndexing({ incremental: false })`, then start polling immediately
    - On error: set `triggerError = error.message`, set `triggering=false`
    - On success: set `triggering=false` (polling will update the UI)

    Progress display:
    - Show stage badge: `walking | parsing | embedding | storing | done` in different colors
    - Progress bar: `(filesProcessed / filesTotal) * 100` for file progress (or `chunksIndexed / chunksTotal` for chunk progress) — whichever is non-zero
    - Show `currentFile` truncated to 50 chars if present
    - Show start time, completion time if available
    - On `error !== null`: show red error banner

    ```tsx
    import { useEffect, useRef, useState } from 'react';
    import { DashboardClient, ApiIndexingStatus } from '../api/client.ts';

    const STAGE_COLORS: Record<string, string> = {
      walking:   'bg-yellow-900/50 text-yellow-300',
      parsing:   'bg-blue-900/50 text-blue-300',
      embedding: 'bg-purple-900/50 text-purple-300',
      storing:   'bg-indigo-900/50 text-indigo-300',
      done:      'bg-green-900/50 text-green-300',
    };

    export default function IndexingPanel() {
      const [status, setStatus] = useState<ApiIndexingStatus | null>(null);
      const [loading, setLoading] = useState(true);
      const [triggering, setTriggering] = useState(false);
      const [triggerError, setTriggerError] = useState<string | null>(null);
      const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);

      function startPolling() {
        if (intervalRef.current) return;
        intervalRef.current = setInterval(() => {
          DashboardClient.getIndexingStatus().then(s => {
            setStatus(s);
            if (!s.running && intervalRef.current) {
              clearInterval(intervalRef.current);
              intervalRef.current = null;
            }
          }).catch(() => {});
        }, 2000);
      }

      useEffect(() => {
        DashboardClient.getIndexingStatus()
          .then(s => {
            setStatus(s);
            if (s.running) startPolling();
          })
          .catch(() => {})
          .finally(() => setLoading(false));

        return () => {
          if (intervalRef.current) clearInterval(intervalRef.current);
        };
      }, []);

      async function handleTrigger() {
        setTriggering(true);
        setTriggerError(null);
        try {
          await DashboardClient.triggerIndexing({ incremental: false });
          startPolling();
        } catch (e: unknown) {
          setTriggerError(e instanceof Error ? e.message : 'Failed to trigger indexing');
        } finally {
          setTriggering(false);
        }
      }

      const isRunning = status?.running ?? false;
      const pct = status && status.filesTotal > 0
        ? Math.round((status.filesProcessed / status.filesTotal) * 100)
        : status && status.chunksTotal > 0
        ? Math.round((status.chunksIndexed / status.chunksTotal) * 100)
        : 0;

      return (
        <div className="bg-gray-900 rounded-xl border border-gray-800 p-5">
    ```
