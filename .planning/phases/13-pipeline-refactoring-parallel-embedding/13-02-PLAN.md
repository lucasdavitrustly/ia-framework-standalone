    ```typescript
    const concurrency = config.embeddingConcurrency ?? getDefaultConcurrency(embeddingProvider.name);
    const queue = new PQueue({ concurrency });
    const batchSize = embeddingBatchSize;
    const batches: CodeChunk[][] = [];
    for (let i = 0; i < allChunks.length; i += batchSize) {
      batches.push(allChunks.slice(i, i + batchSize));
    }

    progress.batchesTotal = batches.length;
    progress.batchesCompleted = 0;

    let skippedChunks = 0;
    const vectorPoints: VectorPoint[] = [];

    // Abort integration per D-09: when signal fires, clear queued tasks
    if (signal) {
      signal.addEventListener('abort', () => { queue.clear(); }, { once: true });
    }

    await Promise.all(batches.map((batch, batchIndex) =>
      queue.add(async () => {
        if (signal?.aborted) return; // Skip if already aborted

        try {
          await embedBatchWithRetry(batch, embeddingProvider, vectorPoints);
        } catch {
          // Per D-07: retry once failed, skip batch
          skippedChunks += batch.length;
          logger.warn({ batchIndex, batchSize: batch.length }, 'Embedding batch skipped after retry failure');
        }

        progress.batchesCompleted = (progress.batchesCompleted ?? 0) + 1;
        progress.chunksIndexed += batch.length - (skippedChunks > 0 ? batch.length : 0);
        onProgress?.({ ...progress });
      })
    ));
    ```

    d) **embedBatchWithRetry helper** per D-07 and D-15:
    ```typescript
    async function embedBatchWithRetry(
      batch: CodeChunk[],
      provider: EmbeddingProvider,
      results: VectorPoint[],
      maxRetries = 1,
    ): Promise<void> {
      let lastError: unknown;
      for (let attempt = 0; attempt <= maxRetries; attempt++) {
        try {
          const textDescriptions = batch.map(c => chunkToDescription(c));
          const codeTexts = batch.map(c => c.content);
          // Per D-15: text and code embeddings run in parallel
          const [textVectors, codeVectors] = await Promise.all([
            provider.embedBatch(textDescriptions),
            provider.embedBatch(codeTexts),
          ]);
          // Build VectorPoints from results
          for (let j = 0; j < batch.length; j++) {
            const chunk = batch[j]!;
            results.push({
              id: chunk.id,
              textVector: textVectors[j]!,
              codeVector: codeVectors[j]!,
              sparseVector: computeTermFrequencySparseVector(chunk.content),
              payload: { /* same payload construction as current */ },
            });
          }
          return; // Success
        } catch (err) {
          lastError = err;
          if (attempt < maxRetries) {
            logger.warn({ attempt, err }, 'Embedding batch failed — retrying');
          }
        }
      }
      throw lastError; // All retries exhausted
    }
    ```
