### Task 1: Add `validateCorpus()` to corpus.ts (TDD)

**File**: `packages/benchmark/src/corpus.ts` and `packages/benchmark/src/corpus.test.ts`

Write tests first, then implement.

**Tests to write** (minimum 7):

1. Empty corpus returns no errors (edge case — valid)
2. Valid corpus with all files existing returns no errors
3. Duplicate `id` values produce an error entry
4. Empty `query` string produces an error entry
5. `expectedFiles` entry referencing a non-existent path produces an error entry
6. Queries with empty `expectedFiles` array are valid (negative queries)
7. Multiple errors are returned together (not short-circuit on first failure)

**Implementation to add to `corpus.ts`**:

```typescript
import * as fs from 'node:fs/promises';
import * as path from 'node:path';

export interface CorpusValidationError {
  queryId: string;
  error: string;
}

/**
 * Validate a loaded corpus against the project root.
 * Checks: unique IDs, non-empty query text, file existence for non-empty expectedFiles.
 * Empty expectedFiles arrays are valid (negative test queries).
 * Returns all errors found (empty array = valid corpus).
 */
export async function validateCorpus(
  queries: BenchmarkQuery[],
  projectRoot: string,
): Promise<CorpusValidationError[]> {
  const errors: CorpusValidationError[] = [];
  const seenIds = new Set<string>();

  for (const q of queries) {
    if (seenIds.has(q.id)) {
      errors.push({ queryId: q.id, error: `Duplicate query ID: "${q.id}"` });
    }
    seenIds.add(q.id);

    if (!q.query.trim()) {
      errors.push({ queryId: q.id, error: 'Query text is empty' });
    }

    // Skip file-existence check for negative queries (expectedFiles=[])
    for (const relPath of q.expectedFiles) {
      const absPath = path.join(projectRoot, relPath);
      try {
        await fs.access(absPath);
      } catch {
        errors.push({ queryId: q.id, error: `Expected file not found: ${relPath}` });
      }
    }
  }

  return errors;
}
```

Export `validateCorpus` and `CorpusValidationError` from `packages/benchmark/src/index.ts`.
