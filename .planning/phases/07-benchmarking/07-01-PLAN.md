    ## Step 3 — BenchmarkStore (TDD: RED → GREEN)

    Write `benchmark-store.test.ts` first (all 6 tests FAILING). Then implement `benchmark-store.ts`.

    Schema:
    ```sql
    CREATE TABLE IF NOT EXISTS benchmark_runs (
      run_id         TEXT PRIMARY KEY,
      label          TEXT,
      is_baseline    INTEGER NOT NULL,    -- 0 or 1
      config_snapshot TEXT NOT NULL,      -- JSON string
      hit_at_k       REAL NOT NULL,
      mrr            REAL NOT NULL,
      avg_latency_ms REAL NOT NULL,
      timestamp      TEXT NOT NULL
    );

    CREATE TABLE IF NOT EXISTS benchmark_query_results (
      id              TEXT PRIMARY KEY,   -- UUID
      run_id          TEXT NOT NULL REFERENCES benchmark_runs(run_id),
      query_id        TEXT NOT NULL,
      query           TEXT NOT NULL,
      expected_files  TEXT NOT NULL,      -- JSON array
      retrieved_files TEXT NOT NULL,      -- JSON array
      hit             INTEGER NOT NULL,   -- 0 or 1
      rank            REAL NOT NULL,      -- Infinity stored as 1e308
      reciprocal_rank REAL NOT NULL,
      latency_ms      INTEGER NOT NULL
    );
    CREATE INDEX IF NOT EXISTS idx_bqr_run_id ON benchmark_query_results(run_id);
    CREATE INDEX IF NOT EXISTS idx_br_timestamp ON benchmark_runs(timestamp);
    ```

    Methods:
    - `saveRun(run: BenchmarkRun): void` — inserts benchmark_runs row + N benchmark_query_results rows in a transaction
    - `getRunById(runId: string): BenchmarkRun | null`
    - `listRuns(limit?: number): BenchmarkRun[]` — newest first, default limit=20
    - `close(): void`

    Note: `Infinity` cannot be stored in SQLite REAL — store as `1e308` and restore via: `rank === 1e308 ? Infinity : rank`.

    ## Step 4 — BenchmarkRunner (TDD: RED → GREEN)

    Write `benchmark-runner.test.ts` first (all 7 tests FAILING). Then implement `benchmark-runner.ts`.

    ```typescript
    export class BenchmarkRunner {
      constructor(
        private readonly searchEngine: { search(q: RetrievalQuery): Promise<RetrievalResult[]> },
        private readonly config: BenchmarkConfig,
      ) {}

      async run(queries: BenchmarkQuery[], configSnapshot: ConfigSnapshot): Promise<BenchmarkRun>
    }
    ```

    Algorithm for each query:
    1. `const start = Date.now()`
    2. `const results = await searchEngine.search({ text: q.query, limit: config.topK, scoreThreshold: config.scoreThreshold })`
    3. `const retrievedFiles = results.map(r => r.relativePath)`
    4. `const hit = q.expectedFiles.some(f => retrievedFiles.includes(f))`
    5. `const rank = hit ? (retrievedFiles.findIndex(f => q.expectedFiles.includes(f)) + 1) : Infinity`
    6. `reciprocalRank = hit ? 1 / rank : 0`
    7. `latencyMs = Date.now() - start`

    Aggregate: `hitAtK = hits/total`, `mrr = sum(reciprocalRank)/total`, `avgLatencyMs = sum(latency)/total`

    `isBaseline = config.scoreThreshold > 1.0`
