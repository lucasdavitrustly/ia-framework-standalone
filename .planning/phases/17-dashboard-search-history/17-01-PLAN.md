    ```typescript
    // GET /history — list history entries with filters
    app.get('/history', (c) => {
      if (!deps.searchHistoryStore) return c.json({ error: 'History not available' }, 503);
      const params = {
        q: c.req.query('q') || undefined,
        collection: c.req.query('collection') || undefined,
        dateFrom: c.req.query('dateFrom') || undefined,
        dateTo: c.req.query('dateTo') || undefined,
        minScore: c.req.query('minScore') ? parseFloat(c.req.query('minScore')!) : undefined,
        sortBy: c.req.query('sortBy') as any || undefined,
        sortOrder: c.req.query('sortOrder') as 'asc' | 'desc' | undefined,
        page: c.req.query('page') ? parseInt(c.req.query('page')!, 10) : undefined,
        pageSize: c.req.query('pageSize') ? parseInt(c.req.query('pageSize')!, 10) : undefined,
      };
      const result = deps.searchHistoryStore.list(params);
      return c.json(result);
    });

    // GET /history/:id — get single entry
    app.get('/history/:id', (c) => {
      if (!deps.searchHistoryStore) return c.json({ error: 'History not available' }, 503);
      const entry = deps.searchHistoryStore.getById(c.req.param('id'));
      if (!entry) return c.json({ error: 'Not found' }, 404);
      return c.json(entry);
    });

    // POST /history/:id/replay — re-execute saved query (per D-05, D-06, D-07)
    app.post('/history/:id/replay', async (c) => {
      if (!deps.searchHistoryStore) return c.json({ error: 'History not available' }, 503);
      const entry = deps.searchHistoryStore.getById(c.req.param('id'));
      if (!entry) return c.json({ error: 'Not found' }, 404);

      // Re-execute with original parameters
      const retrievalQuery: RetrievalQuery = {
        text: entry.query,
        limit: entry.params.limit,
        scoreThreshold: entry.params.scoreThreshold,
      };

      const collections = entry.params.collections;
      const wantAst = collections.includes('ast');
      const wantLlm = collections.includes('llm');
      const wantBoth = collections.includes('both') || (wantAst && wantLlm);

      const start = Date.now();
      let rawResults;
      let collectionWarning: string | undefined;

      try {
        if (wantBoth) {
          rawResults = await deps.multiSearchEngine.search(retrievalQuery);
        } else if (wantAst) {
          const engineEntry = deps.collectionEngines.find(e => e.name === 'ast');
          if (!engineEntry) { collectionWarning = 'AST collection no longer available'; rawResults = []; }
          else rawResults = await engineEntry.engine.search(retrievalQuery);
        } else if (wantLlm) {
          const engineEntry = deps.collectionEngines.find(e => e.name === 'llm');
          if (!engineEntry) { collectionWarning = 'LLM collection no longer available'; rawResults = []; }
          else rawResults = await engineEntry.engine.search(retrievalQuery);
        } else {
          rawResults = await deps.multiSearchEngine.search(retrievalQuery);
        }
      } catch {
        return c.json({ error: 'Replay search failed' }, 500);
      }

      const latencyMs = Date.now() - start;

      // Map same as POST / handler
      const freshResults = rawResults.map(r => ({
        id: r.chunkId, filePath: r.relativePath, startLine: r.startLine, endLine: r.endLine,
        score: r.score, source: r.source ?? 'ast', chunkType: r.chunkType,
        matchedVector: r.matchedVector, content: r.content, description: r.name,
      }));

      const freshMetrics = {
        latencyMs,
        chunksScanned: 0, // simplified — could compute but not critical for diff
        resultsReturned: freshResults.length,
        scoreRange: {
          min: freshResults.length > 0 ? Math.min(...freshResults.map(r => r.score)) : 0,
          max: freshResults.length > 0 ? Math.max(...freshResults.map(r => r.score)) : 0,
        },
      };

      return c.json({
        original: { results: entry.results, metrics: entry.metrics },
        fresh: { results: freshResults, metrics: freshMetrics },
        ...(collectionWarning ? { warning: collectionWarning } : {}),
      });
    });

    // DELETE /history/:id — delete single entry (per D-21)
    app.delete('/history/:id', (c) => {
      if (!deps.searchHistoryStore) return c.json({ error: 'History not available' }, 503);
      const deleted = deps.searchHistoryStore.deleteById(c.req.param('id'));
      if (!deleted) return c.json({ error: 'Not found' }, 404);
      return c.json({ ok: true });
    });

    // DELETE /history — clear all (per D-22)
    app.delete('/history', (c) => {
      if (!deps.searchHistoryStore) return c.json({ error: 'History not available' }, 503);
      deps.searchHistoryStore.clearAll();
      return c.json({ ok: true });
    });
    ```
