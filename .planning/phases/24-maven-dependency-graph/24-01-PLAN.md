    export interface MavenCoordinate {
      groupId: string;
      artifactId: string;
      version: string;
    }

    export interface MavenModuleInfo {
      coordinate: MavenCoordinate;
      /** groupId:artifactId */
      moduleId: string;
      dependencies: string[];       // moduleIds of direct deps
      interfaces: string[];          // Java type FQNs that cross boundaries
      dtos: string[];                // Java type FQNs that are DTOs
      pomPath: string;               // relative path to this module's pom.xml
      sourceRoot: string;            // relative path to src/main/java
    }

    export interface ModuleGraph {
      modules: Map<string, MavenModuleInfo>;
      /** Get direct dependency moduleIds */
      getDirectDeps(moduleId: string): string[];
      /** Get transitive deps up to N hops */
      getTransitiveDeps(moduleId: string, maxHops: number): string[];
      /** Serialize for cache file */
      toJSON(): ModuleGraphJSON;
    }

    export interface ModuleGraphJSON {
      version: 1;
      modules: Record<string, Omit<MavenModuleInfo, 'moduleId'>>;
    }
    ```

    saveModuleGraph(projectRoot: string, graphJson: ModuleGraphJSON): void {
      this.db.prepare(`
        INSERT INTO module_graph (project_root, graph_json, updated_at)
        VALUES (?, ?, ?)
        ON CONFLICT(project_root) DO UPDATE SET graph_json = excluded.graph_json, updated_at = excluded.updated_at
      `).run(projectRoot, JSON.stringify(graphJson), new Date().toISOString());
    }

    getModuleGraph(projectRoot: string): ModuleGraphJSON | null {
      const row = this.db.prepare('SELECT graph_json FROM module_graph WHERE project_root = ?').get(projectRoot);
      return row ? JSON.parse((row as { graph_json: string }).graph_json) : null;
    }
    ```
