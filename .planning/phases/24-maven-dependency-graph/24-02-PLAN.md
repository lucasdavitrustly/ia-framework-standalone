    export function inferMavenModule(
      relativePath: string,
      graph: ModuleGraph,
    ): MavenCoordinate | undefined {
      // Match file path prefix against module sourceRoot paths
      const normalized = relativePath.replace(/\\/g, '/');
      for (const [, info] of graph.modules) {
        if (normalized.startsWith(info.sourceRoot) || normalized.startsWith(info.pomPath.replace('/pom.xml', '/'))) {

        }
      }
      return undefined;
    }

    1. Create `packages/mcp-server/src/tools/get-module-dependencies.ts` (per D-06):

    ```typescript
    import { McpServer } from '@modelcontextprotocol/sdk/server/mcp.js';
    import type { CallToolResult } from '@modelcontextprotocol/sdk/types.js';
    import { z } from 'zod';

    export interface ModuleDepsEngine {
      getModuleGraph(): ModuleGraphJSON | null;
      searchByModule(moduleName: string, limit: number): Promise<SearchResult[]>;
    }

    export interface GetModuleDependenciesInput {
      moduleName: string;
    }

    export async function getModuleDependenciesHandler(
      engine: ModuleDepsEngine,
      input: GetModuleDependenciesInput,
    ): Promise<CallToolResult> {
      const graph = engine.getModuleGraph();
      if (!graph) {
        return { content: [{ type: 'text', text: 'No Maven module graph available. Run indexing on a Maven project first.' }] };
      }

      // Find module — exact match on moduleId, or partial match on artifactId
      const moduleId = findModuleId(graph, input.moduleName);
      if (!moduleId) {
        const available = Object.keys(graph.modules).join(', ');
        return { content: [{ type: 'text', text: `Module "${input.moduleName}" not found. Available: ${available}` }] };
      }

      const info = graph.modules[moduleId];
      const directDeps = info.dependencies;
      const transitiveDeps = getTransitiveDepsFromJSON(graph, moduleId, 2);

      // Fetch top-5 interface/DTO chunks from Qdrant filtered by maven_module
      const interfaceChunks = await engine.searchByModule(moduleId, 5);

      return {
        content: [{
          type: 'text',
          text: formatModuleDeps(moduleId, info, directDeps, transitiveDeps, interfaceChunks),
        }],
      };
    }
    ```

    - `findModuleId`: if input contains ":", exact lookup; else, find first moduleId where artifactId matches (partial, case-insensitive)
    - `getTransitiveDepsFromJSON`: BFS from moduleId, max 2 hops, return unique moduleIds excluding start
    - `formatModuleDeps`: Format as structured text with sections: Module, Direct Dependencies, Transitive Dependencies (2 hops), Key Interface/DTO Chunks (code snippets)
