    Create `packages/mcp-server/src/classification/` directory.

    **query-classifier.ts exports:**
    ```typescript
    export type QueryType = 'keyword' | 'semantic';
    export interface QueryClassification {
      type: QueryType;
      reason: string;
      confidence: number; // 0-1, based on how many signals matched
    }
    export interface ClassificationRules {
      symbolPatterns?: RegExp[];
      lookupVerbs?: string[];
      maxKeywordWords?: number;
    }
    export const DEFAULT_RULES: ClassificationRules = {
      symbolPatterns: [
        /[A-Z][a-z]+[A-Z]/,           // PascalCase with multiple words
        /^[a-z]+[A-Z]/,               // camelCase
        /\w+\.\w+/,                   // dot-notation
        /^[A-Z][A-Za-z]+$/,           // Single PascalCase word
      ],
      lookupVerbs: ['find', 'where', 'locate', 'show', 'get', 'what is'],
      maxKeywordWords: 8,
    };
    export function classifyQuery(query: string, rules?: ClassificationRules): QueryClassification;
    ```

    **Classification logic (in order):**
    1. If query contains quoted substring → keyword (reason: 'quoted-string')
    2. If query matches any symbolPattern → keyword (reason: 'symbol-pascalcase' | 'symbol-camelcase' | 'symbol-dotnotation')
    3. If query starts with a lookup verb AND word count <= maxKeywordWords → keyword (reason: 'lookup-verb')
    4. If query is <=3 words and all words look like identifiers → keyword (reason: 'short-identifier')
    5. Otherwise → semantic (reason: 'default')

    Confidence: 1.0 for quoted strings/symbols, 0.8 for lookup verbs, 0.6 for short-identifier, 0.5 for default.

    **index.ts:** Re-export everything from query-classifier.ts.

    Per D-01: Pattern-based heuristics only, no LLM call.
    Per D-02: Binary output (keyword vs semantic), no hybrid category.
    Per D-03: Standalone module in packages/mcp-server.
